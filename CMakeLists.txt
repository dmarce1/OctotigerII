option(OCTOTIGERII_WITH_HPX "Use distributed HPX field storage and locality schedulers" ON)
option(OCTOTIGERII_WITH_PROFILING "Name HPX tasks and enable APEX regions when available" ON)
option(OCTOTIGERII_BUILD_TESTS "Build the small numerical/example test suite" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
include(GNUInstallDirs)
include(cmake/Problems.cmake)
find_package(Threads REQUIRED)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
find_package(Boost 1.71 REQUIRED)

function(octo_library target)
  add_library(${target} STATIC EXCLUDE_FROM_ALL ${ARGN})
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${PROJECT_BINARY_DIR}/include>
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_compile_features(${target} PUBLIC cxx_std_20)
  target_link_libraries(${target} PUBLIC Boost::headers)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

foreach(module Mesh Physics Hydro Radiation Gravity Subgrid Application)
  add_library(OctotigerII::${module} ALIAS octotigerII${module})
endforeach()

if(OCTOTIGERII_WITH_HPX)
  find_package(HPX 1.11 REQUIRED)
  if(DEFINED HPX_WITH_DISTRIBUTED_RUNTIME AND NOT HPX_WITH_DISTRIBUTED_RUNTIME)
    message(FATAL_ERROR "The HPX build must enable its distributed runtime")
  endif()
  target_compile_definitions(octotigerIISubgrid PRIVATE OCTOTIGERII_WITH_HPX=1)
  target_link_libraries(octotigerIISubgrid PRIVATE HPX::hpx)
  target_compile_definitions(octotigerIIApplication PUBLIC OCTOTIGERII_WITH_HPX=1)
  target_link_libraries(octotigerIIApplication PUBLIC HPX::hpx)
  if(OCTOTIGERII_WITH_PROFILING)
    # Use a separate definition: numerical libraries need profiling, not the
    # distributed-storage API. PUBLIC keeps inline template definitions uniform.
    foreach(module Mesh Physics Hydro Radiation Gravity Subgrid Application)
      target_compile_definitions(octotigerII${module} PUBLIC OCTOTIGERII_PROFILE_HPX=1)
      target_link_libraries(octotigerII${module} PUBLIC HPX::hpx)
    endforeach()
    message(STATUS "Profiling: HPX task annotations enabled; APEX regions follow HPX_HAVE_APEX")
  endif()
  if(TARGET HPX::parcel_coalescing)
    target_link_libraries(octotigerIIApplication PUBLIC HPX::parcel_coalescing)
  endif()
else()
  # Preserve the HPX-free serial build with the same Program_options API.
  find_package(Boost 1.71 REQUIRED COMPONENTS program_options)
  target_link_libraries(octotigerIISubgrid PRIVATE Boost::program_options)
endif()

find_path(Silo_INCLUDE_DIR NAMES silo.h HINTS ${Silo_ROOT} PATH_SUFFIXES include REQUIRED)
find_library(Silo_LIBRARY NAMES siloh5 HINTS ${Silo_ROOT} PATH_SUFFIXES lib lib64 REQUIRED)
find_package(HDF5 REQUIRED COMPONENTS C)
find_package(ZLIB REQUIRED)
target_include_directories(octotigerIIApplication PRIVATE ${Silo_INCLUDE_DIR} ${HDF5_INCLUDE_DIRS})
target_link_libraries(octotigerIIApplication PRIVATE ${Silo_LIBRARY} ${HDF5_LIBRARIES} ZLIB::ZLIB ${CMAKE_DL_LIBS})

add_executable(octotigerII src/main.cpp)
target_link_libraries(octotigerII PRIVATE octotigerIIApplication)
set_target_properties(octotigerII PROPERTIES OUTPUT_NAME "${OCTOTIGERII_EXECUTABLE}")
if(OCTOTIGERII_WITH_HPX)
  # HPX::hpx supplies the runtime; this executable calls hpx::init itself.
  target_link_libraries(octotigerII PRIVATE HPX::hpx)
endif()
install(TARGETS octotigerII RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${PROBLEM_DIRECTORY}/inputs" DESTINATION ${CMAKE_INSTALL_DATADIR}/OctotigerII/${OCTOTIGERII_PROBLEM})

  enable_testing()
  add_subdirectory(tests)
endif()
