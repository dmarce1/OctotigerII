cmake_minimum_required(VERSION 3.22)
project(OctotigerII VERSION 0.1.0 LANGUAGES C CXX)

option(OCTOTIGERII_WITH_HPX "Run blocks as distributed HPX components" ON)
option(OCTOTIGERII_WITH_SILO "Enable Silo/HDF5 visualization output" ON)
option(OCTOTIGERII_BUILD_TESTS "Build the small numerical/example test suite" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INSTALL_RPATH_USE_LINK_PATH TRUE)
include(GNUInstallDirs)
find_package(Threads REQUIRED)

function(octo_library target)
  add_library(${target} STATIC ${ARGN})
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_compile_features(${target} PUBLIC cxx_std_20)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

octo_library(octotigerIIMesh src/mesh.cpp)
octo_library(octotigerIIPhysics src/physics/finiteVolume.cpp)
target_link_libraries(octotigerIIPhysics PUBLIC octotigerIIMesh)
octo_library(octotigerIIHydro src/hydro/hydroSystem.cpp)
target_link_libraries(octotigerIIHydro PUBLIC octotigerIIPhysics)
octo_library(octotigerIIRadiation src/radiation/radiationTransport.cpp)
target_link_libraries(octotigerIIRadiation PUBLIC octotigerIIPhysics)
octo_library(octotigerIIGravity src/gravity/gravityFields.cpp src/gravity/diagonal/fmm.cpp src/gravity/solver.cpp)
target_link_libraries(octotigerIIGravity PUBLIC octotigerIIMesh Threads::Threads)
octo_library(octotigerIISubgrid src/subgrid.cpp src/config.cpp src/problems.cpp)
target_link_libraries(octotigerIISubgrid PUBLIC octotigerIIHydro octotigerIIRadiation octotigerIIGravity)
octo_library(octotigerIIApplication src/runtime.cpp src/simulation.cpp src/output.cpp)
target_link_libraries(octotigerIIApplication PUBLIC octotigerIISubgrid)

foreach(module Mesh Physics Hydro Radiation Gravity Subgrid Application)
  add_library(OctotigerII::${module} ALIAS octotigerII${module})
endforeach()

if(OCTOTIGERII_WITH_HPX)
  find_package(HPX 1.11 REQUIRED)
  if(DEFINED HPX_WITH_DISTRIBUTED_RUNTIME AND NOT HPX_WITH_DISTRIBUTED_RUNTIME)
    message(FATAL_ERROR "The HPX build must enable its distributed runtime for Subgrid components")
  endif()
  target_compile_definitions(octotigerIIApplication PUBLIC OCTOTIGERII_WITH_HPX=1)
  target_link_libraries(octotigerIIApplication PUBLIC HPX::hpx)
endif()

if(OCTOTIGERII_WITH_SILO)
  find_path(Silo_INCLUDE_DIR NAMES silo.h HINTS ${Silo_ROOT} PATH_SUFFIXES include REQUIRED)
  find_library(Silo_LIBRARY NAMES siloh5 HINTS ${Silo_ROOT} PATH_SUFFIXES lib lib64 REQUIRED)
  find_package(HDF5 REQUIRED COMPONENTS C)
  find_package(ZLIB REQUIRED)
  target_include_directories(octotigerIIApplication PRIVATE ${Silo_INCLUDE_DIR} ${HDF5_INCLUDE_DIRS})
  target_link_libraries(octotigerIIApplication PRIVATE ${Silo_LIBRARY} ${HDF5_LIBRARIES} ZLIB::ZLIB ${CMAKE_DL_LIBS})
  target_compile_definitions(octotigerIISubgrid PUBLIC OCTOTIGERII_WITH_SILO=1)
endif()

add_executable(octotigerII src/main.cpp)
target_link_libraries(octotigerII PRIVATE octotigerIIApplication)
if(OCTOTIGERII_WITH_HPX)
  # HPX::hpx supplies the runtime; this executable calls hpx::init itself.
  target_link_libraries(octotigerII PRIVATE HPX::hpx)
endif()
install(TARGETS octotigerII RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/OctotigerII/examples)

if(OCTOTIGERII_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
