function(octo_library target)
  add_library(${target} STATIC EXCLUDE_FROM_ALL ${ARGN})
  target_include_directories(${target} PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>
    $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  target_compile_features(${target} PUBLIC cxx_std_20)
  target_link_libraries(${target} PUBLIC Boost::headers)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

if(OCTOTIGERII_WITH_HPX)
  find_package(HPX 1.11 REQUIRED)
  if(DEFINED HPX_WITH_DISTRIBUTED_RUNTIME AND NOT HPX_WITH_DISTRIBUTED_RUNTIME)
    message(FATAL_ERROR "The HPX build must enable its distributed runtime")
  endif()
  target_compile_definitions(octotigerIISubgrid${suffix} PRIVATE OCTOTIGERII_WITH_HPX=1)
  target_link_libraries(octotigerIISubgrid${suffix} PRIVATE HPX::hpx)
  target_compile_definitions(octotigerIIGravity${suffix} PRIVATE OCTOTIGERII_WITH_HPX=1)
  target_link_libraries(octotigerIIGravity${suffix} PRIVATE HPX::hpx)
  target_compile_definitions(octotigerIIApplication${suffix} PUBLIC OCTOTIGERII_WITH_HPX=1)
  target_link_libraries(octotigerIIApplication${suffix} PUBLIC HPX::hpx)
  if(OCTOTIGERII_WITH_PROFILING)
    # Use a separate definition: numerical libraries need profiling, not the
    # distributed-storage API. PUBLIC keeps inline template definitions uniform.
    foreach(module Mesh Physics Hydro Radiation Gravity Composition Subgrid Refinement Amr Application)
      if(TARGET octotigerII${module}${suffix})
      target_compile_definitions(octotigerII${module}${suffix} PUBLIC OCTOTIGERII_PROFILE_HPX=1)
      target_link_libraries(octotigerII${module}${suffix} PUBLIC HPX::hpx)
      endif()
    endforeach()
    message(STATUS "Profiling: HPX task annotations enabled; APEX regions follow HPX_HAVE_APEX")
  endif()
  if(TARGET HPX::parcel_coalescing)
    target_link_libraries(octotigerIIApplication${suffix} PUBLIC HPX::parcel_coalescing)
  endif()
else()
  # Preserve the HPX-free serial build with the same Program_options API.
  find_package(Boost 1.71 REQUIRED COMPONENTS program_options)
  target_link_libraries(octotigerIISubgrid${suffix} PRIVATE Boost::program_options)
endif()
