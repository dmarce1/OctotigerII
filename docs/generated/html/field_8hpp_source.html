24class PartitionSet {
25public:
26
27#ifdef OCTOTIGERII_WITH_HPX
28 using Partition = hpx::id_type;
29#else
30 using Partition = std::shared_ptr<StoragePartition>;

33 explicit PartitionSet(std::vector<Locality> owners)
34 : state_(std::make_shared<State>()) {
35 if (owners.empty()) throw std::invalid_argument("Storage needs at least one partition");
36 state_->owners = std::move(owners);
37 for (auto const& owner : state_->owners) {
38#ifdef OCTOTIGERII_WITH_HPX
39 state_->partitions.push_back(hpx::new_<StoragePartition>(owner).get());
40#else
41 (void) owner;
42 state_->partitions.push_back(std::make_shared<StoragePartition>());
43#endif
44 }
45 }
46
48 std::vector<Locality> const& owners() const {
49 return state_->owners;
50 }
51
53 std::vector<Partition> const& partitions() const {
54 return state_->partitions;
55 }
56
58 FieldId newField() const {
59 std::lock_guard lock(state_->mutex);
60 if (state_->next == std::numeric_limits<FieldId>::max()) throw std::overflow_error("Storage field identity exhausted");
61 return state_->next++;
62 }

64private:

86template <typename T>
87class FieldHandle {
88public:
89
90 Layout layout;
91 std::vector<Locality> owners;
92 std::vector<PartitionSet::Partition> partitions;
93 FieldId id = 0;
94 unsigned banks = 0;
95
97 bool local(Range const& range) const {
98#ifdef OCTOTIGERII_WITH_HPX
99 return owners.at(range.partition) == hpx::find_here();
100#else
101 (void) range;
102 return true;
103#endif
104 }
105
107 Future<Buffer<T>> read(Range range, unsigned bank) const {
108 validate(range, bank);
109#ifdef OCTOTIGERII_WITH_HPX
110 if (local(range)) return hpx::make_ready_future(readField<T>(partitions.at(range.partition), id, bank, range.offset, range.count));
111 return hpx::async<ReadAction<T>>(owners.at(range.partition), partitions.at(range.partition), id, bank, range.offset, range.count);
112#else
113 return Future<Buffer<T>>(partitions.at(range.partition)->template read<T>(id, bank, range.offset, range.count));
114#endif
115 }
116
118 Buffer<T> output(Range range, unsigned bank) const {
119 validate(range, bank);
120 return local(range) ? read(range, bank).get() : Buffer<T>(range.count);
121 }
122
124 void commit(Range range, unsigned bank, Buffer<T> const& output) const {
125 validate(range, bank);
126 if (output.size() != range.count) throw std::invalid_argument("Field commit size mismatch");
127#ifdef OCTOTIGERII_WITH_HPX
128 if (!local(range))
129 hpx::async<WriteAction<T>>(owners.at(range.partition), partitions.at(range.partition), id, bank, range.offset, output).get();
130 else
131#endif
132 {
133 // Also accept caller-owned buffers, while leaving direct local
134 // output views untouched. Both paths validate the field identity.
135 auto destination = read(range, bank).get();
136 if (destination.data() != output.data() && range.count) std::copy_n(output.data(), range.count, destination.data());
137 }
138 }
139
141 template <typename Archive>
142 void serialize(Archive& archive, unsigned) {
143 archive & layout & owners & partitions & id & banks;
144 }
145
146private:
147
149 void validate(Range const& range, unsigned bank) const {
150 layout.validate(range);
151 if (!id || bank >= banks) throw std::out_of_range("Invalid field identity or stage bank");
152 }
153};

162template <typename T>
163class Field {
164public:
165
166 Field(Layout const& layout, std::vector<Locality> const& owners, unsigned banks = 2, std::string name = {})
167 : Field(layout, PartitionSet(owners), banks, std::move(name)) {}
168
169 Field(Layout const& layout, PartitionSet const& store, unsigned banks = 2, std::string name = {}) {
170 if (store.owners().size() != layout.partitionCount() || banks == 0) throw std::invalid_argument("Invalid field placement or bank count");
171 handle_.layout = layout.addressSpace();
172 handle_.owners = store.owners();
173 handle_.partitions = store.partitions();
174 handle_.id = store.newField();
175 handle_.banks = banks;
176 try {
177 for (std::size_t i = 0; i < handle_.partitions.size(); ++i) {
178#ifdef OCTOTIGERII_WITH_HPX
179 hpx::async<CreateAction<T>>(handle_.owners[i], handle_.partitions[i], handle_.id, layout.capacity(i), banks, name).get();
180#else
181 handle_.partitions[i]->template create<T>(handle_.id, layout.capacity(i), banks, name);
182#endif
183 }
184 } catch (...) {
185 auto error = std::current_exception();
186 try {
187 retire();
188 } catch (...) {
189 }
190 std::rethrow_exception(error);
191 }
192 }
193
195 FieldHandle<T> const& handle() const {
196 return handle_;
197 }
198
201 void retire() {
202 std::exception_ptr error;
203 for (std::size_t i = 0; i < handle_.partitions.size(); ++i) {
204 try {
205#ifdef OCTOTIGERII_WITH_HPX
206 hpx::async<RetireAction>(handle_.owners[i], handle_.partitions[i], handle_.id).get();
207#else
208 handle_.partitions[i]->retire(handle_.id);
209#endif
210 } catch (...) {
211 if (!error) error = std::current_exception();
212 }
213 }
214 if (error) std::rethrow_exception(error);
215 }
216
217private:
218
219 FieldHandle<T> handle_;
220};
