30HaloPlan makeHaloPlan(Config const& config_, std::vector<Subgrid> const& blocks_, std::size_t index) {
31 auto const& block = blocks_.at(index);
32 int const n = 1 << config_.mesh.level;
33 int const globalCells = n * config_.mesh.cells;
34 mesh::MeshLayout const padded(config_.mesh.cells, 2);
35 HaloPlan plan;
36 plan.ghostIndices.resize(padded.cellCount(), std::numeric_limits<std::size_t>::max());
37 // Group identical source cells and then join adjacent addresses. This is
38 // a geometry plan; transport message coalescing remains HPX's responsibility.
39 using Address = std::pair<std::size_t, std::size_t>;
40 std::map<Address, std::vector<std::size_t>> donors;
41 auto const extents = padded.extents();
42 mesh::forEachCoordinate(extents, [&](mesh::Coordinates cell) {
43 if (padded.isInterior(cell)) return;
44 auto const ghost = plan.ghostCount++;
45 plan.ghostIndices[padded.index(cell)] = ghost;
46 mesh::Coordinates sourceBlock{};
47 for (int axis = 0; axis < ndim; ++axis) {
48 int coordinate = block.location.coordinates[axis] * config_.mesh.cells + cell[axis] - 2;
49 if (config_.mesh.periodic)
50 coordinate = (coordinate % globalCells + globalCells) % globalCells;
51 else
52 coordinate = std::clamp(coordinate, 0, globalCells - 1);
53 sourceBlock[axis] = coordinate / config_.mesh.cells;
54 cell[axis] = coordinate % config_.mesh.cells;
55 }
56 auto const sourceId = mesh::linearIndex(sourceBlock, mesh::filledCoordinates(n));
57 auto const& source = blocks_.at(sourceId);
58 auto const offset = source.interior.offset + source.layout.index(cell);
59 donors[{source.interior.partition, offset}].push_back(ghost);
60 });
61 for (auto const& [address, destinations] : donors) {
62 auto const [partition, offset] = address;
63 if (plan.reads.empty() || plan.reads.back().range.partition != partition || plan.reads.back().range.offset + plan.reads.back().range.count != offset)
64 plan.reads.push_back({{partition, offset, 0}, {}});
65 auto& read = plan.reads.back();
66 for (auto const destination : destinations)
67 read.copies.push_back({read.range.count, destination});
68 ++read.range.count;
69 }
70 return plan;
71}
