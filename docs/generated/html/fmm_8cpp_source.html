48// Golub-Welsch is unnecessary at these tiny orders: Newton iteration on L_n.
49std::vector<std::pair<double, double>> laguerre(int n) {
50 using std::abs;
51
52 std::vector<std::pair<double, double>> q;
53 double z = 0;
54 for (int i = 0; i < n; ++i) {
55 if (i == 0)
56 z = 3.0 / (1 + 2.4 * n);
57 else if (i == 1)
58 z += 15.0 / (1 + 2.5 * n);
59 else {
60 const double a = i - 1;
61 z += (1 + 2.55 * a) / (1.9 * a) * (z - q[i - 2].first);
62 }
63 double prev = 0, deriv = 0;
64 bool converged = false;
65 for (int it = 0; it < 50; ++it) {
66 double l = 1;
67 prev = 0;
68 for (int j = 1; j <= n; ++j) {
69 const double old = prev;
70 prev = l;
71 l = ((2 * j - 1 - z) * prev - (j - 1) * old) / j;
72 }
73 deriv = n * (l - prev) / z;
74 const double dz = l / deriv;
75 z -= dz;
76 if (abs(dz) < 2e-15 * (1 + abs(z))) {
77 converged = true;
78 break;
79 }
80 }
81 if (!converged) throw std::runtime_error("diagonal FMM Laguerre quadrature failed");
82 // Reevaluate derivative at the converged root.
83 double l = 1;
84 prev = 0;
85 for (int j = 1; j <= n; ++j) {
86 const double old = prev;
87 prev = l;
88 l = ((2 * j - 1 - z) * prev - (j - 1) * old) / j;
89 }
90 deriv = n * (l - prev) / z;
91 q.emplace_back(z, 1 / (z * deriv * deriv));
92 }
93 return q;
94}

104const std::vector<Index>& indices(int p) {
105 coefficientCount(p);
106 static const auto tables = [] {
107 std::array<std::vector<Index>, 11> result;
108 for (int p = 0; p <= 10; ++p)
109 for (int n = 0; n <= p; ++n) {
110 for (int x = 0; x <= n; ++x)
111 result[p].push_back({x, n - x, 0});
112 for (int x = 0; x < n; ++x)
113 result[p].push_back({x, n - 1 - x, 1});
114 }
115 return result;
116 }();
117 return tables[p];
118}

120Coefficients point(double mass, const Vector& s, int p) {
121 Coefficients a(coefficientCount(p), 0);
122 for (int n = 0; n <= p; ++n)
123 for (int x = 0; x <= n; ++x)
124 for (int y = 0; y <= n - x; ++y) {
125 const int z = n - x - y;
126 reduce(a, x, y, z, mass * power(s[0], x) * power(s[1], y) * power(s[2], z) / (factorial(x) * factorial(y) * factorial(z)));
127 }
128 return a;
129}

131Coefficients shiftMultipole(const Coefficients& a, const Vector& s, double ratio, int p) {
132 const auto e = point(1, s, p);
133 const auto& ix = indices(p);
134 if (a.size() != 1 && a.size() != ix.size()) throw std::invalid_argument("invalid multipole size");
135 Coefficients b(ix.size(), 0);
136 for (std::size_t i = 0; i < a.size(); ++i)
137 for (std::size_t j = 0; j < ix.size(); ++j) {
138 if (ix[i].degree() + ix[j].degree() > p) continue;
139 reduce(b, ix[i].x + ix[j].x, ix[i].y + ix[j].y, ix[i].z + ix[j].z, a[i] * power(ratio, ix[i].degree()) * e[j]);
140 }
141 return b;
142}

149Coefficients shiftLocal(const Coefficients& l, const Vector& s, double ratio, int p) {
150 const auto& ix = indices(p);
151 if (l.size() != ix.size()) throw std::invalid_argument("invalid local size");
152 Coefficients b(ix.size(), 0);
153 for (std::size_t i = 0; i < ix.size(); ++i) {
154 const auto a = ix[i];
155 for (int n = 0; n <= p - a.degree(); ++n)
156 for (int x = 0; x <= n; ++x)
157 for (int y = 0; y <= n - x; ++y) {
158 const int z = n - x - y;
159 b[i] += derivative(l, a.x + x, a.y + y, a.z + z) * power(s[0], x) * power(s[1], y) * power(s[2], z) /
160 (factorial(x) * factorial(y) * factorial(z));
161 }
162 b[i] *= power(ratio, a.degree());
163 }
164 return b;
165}

172Operator::Operator(int p, Offset r)
173 : count_(coefficientCount(p)) {
174 using std::abs;
175 using std::cos;
176 using std::hypot;
177 using std::sin;
178
179 if (p < 1) throw std::invalid_argument("diagonal FMM M2L order must be 1..10");
180 const double radius = hypot(double(r[0]), double(r[1]), double(r[2]));
181 if (radius == 0) throw std::invalid_argument("zero M2L separation");
182 Vector e{r[0] / radius, r[1] / radius, r[2] / radius};
183 int a = 0;
184 for (int d = 1; d < 3; ++d)
185 if (abs(e[d]) < abs(e[a])) a = d;
186 Vector u{};
187 u[a] = 1;
188 const double dot = e[a];
189 for (int d = 0; d < 3; ++d)
190 u[d] -= dot * e[d];
191 const double norm = hypot(u[0], u[1], u[2]);
192 for (auto& v : u)
193 v /= norm;
194 Vector v{e[1] * u[2] - e[2] * u[1], e[2] * u[0] - e[0] * u[2], e[0] * u[1] - e[1] * u[0]};
195 // Source degree p + local derivative degree p: radial degree <=2p,
196 // angular Fourier degree <=2p. Both quadratures are exact at these sizes.
197 const int angles = 2 * p + 1;
198 for (auto [t, w] : laguerre(p + 1))
199 for (int j = 0; j < angles; ++j) {
200 const double angle = 2 * pi * j / angles;
201 std::array<std::complex<double>, 3> k;
202 for (int d = 0; d < 3; ++d)
203 k[d] = t / radius * std::complex<double>(-e[d], u[d] * cos(angle) + v[d] * sin(angle));
204 // exp(k.R)=exp(-t) is the diagonal translation. Its exp(-t)
205 // factor is already in the Gauss-Laguerre weight w.
206 diagonal_.push_back(-w / (radius * angles));
207 for (const auto b : indices(p)) {
208 std::complex<double> z = 1;
209 for (int n = 0; n < b.x; ++n)
210 z *= k[0];
211 for (int n = 0; n < b.y; ++n)
212 z *= k[1];
213 for (int n = 0; n < b.z; ++n)
214 z *= k[2];
215 fromWave_.push_back(z);
216 toWave_.push_back((b.degree() % 2 ? -1.0 : 1.0) * z);
217 }
218 }

221void Operator::add(Coefficients& l, const Coefficients& m, double h) const {
222 using std::isfinite;
223
224 if (!(h > 0) || !isfinite(h) || (m.size() != 1 && m.size() != std::size_t(count_)) || (l.size() != 4 && l.size() != std::size_t(count_)))
225 throw std::invalid_argument("invalid M2L data");
226 for (std::size_t q = 0; q < diagonal_.size(); ++q) {
227 std::complex<double> wave = 0;
228 const auto off = q * count_;
229 for (std::size_t i = 0; i < m.size(); ++i)
230 wave += toWave_[off + i] * m[i];
231 wave *= diagonal_[q] / h;
232 for (std::size_t i = 0; i < l.size(); ++i)
233 l[i] += (fromWave_[off + i] * wave).real();
234 }
235}

237std::shared_ptr<const Operator> getOperator(int p, Offset r) {
238 const std::array<int, 4> key{p, r[0], r[1], r[2]};
239 {
240 std::shared_lock lock(cacheMutex);
241 const auto it = cache.find(key);
242 if (it != cache.end()) return it->second;
243 }
244 // Short construction, no suspension or HPX calls under the lock. Build once.
245 std::unique_lock lock(cacheMutex);
246 auto it = cache.find(key);
247 if (it != cache.end()) return it->second;
248 auto op = std::make_shared<const Operator>(p, r);
249 cache.emplace(key, op);
250 return op;
251}

258void addDirect(Coefficients& l, double m, Offset r, double h) {
259 using std::hypot;
260
261 const double distance = hypot(double(r[0]), double(r[1]), double(r[2]));
262 if (distance == 0 || !(h > 0)) throw std::invalid_argument("invalid P2P separation");
263 l.at(0) -= m / (h * distance);
264 const double f = m / (h * distance * distance * distance);
265 l.at(index(1, 0, 0)) += f * r[0];
266 l.at(index(0, 1, 0)) += f * r[1];
267 l.at(index(0, 0, 1)) += f * r[2];
268}
