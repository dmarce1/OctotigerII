69std::size_t MeshLayout::index(Coordinates const& storageCoordinates) const {
70 std::size_t result = 0;
71 for (int axis = 0; axis < ndim; ++axis) {
72 if (storageCoordinates[axis] < 0 || storageCoordinates[axis] >= extents_[axis]) {
73 throw std::out_of_range("Mesh storage coordinate is outside the allocated extent");
74 }
75 result += static_cast<std::size_t>(storageCoordinates[axis]) * strides_[axis];

102bool MeshLayout::isInterior(Coordinates const& storageCoordinates) const {
103 for (int axis = 0; axis < ndim; ++axis) {
104 if (storageCoordinates[axis] < 0 || storageCoordinates[axis] >= extents_[axis]) {
105 return false;
106 }
107 if (storageCoordinates[axis] < ghostWidth_ || storageCoordinates[axis] >= ghostWidth_ + cellsPerActiveDimension_) {
108 return false;
109 }
110 }
111 return true;

125std::size_t MeshLayout::faceIndex(int normal, Coordinates const& faceCoordinates) const {
126 Coordinates const dimensions = faceExtents(normal);
127 std::size_t stride = 1;
128 std::size_t result = 0;
129 for (int axis = 0; axis < ndim; ++axis) {
130 if (faceCoordinates[axis] < 0 || faceCoordinates[axis] >= dimensions[axis]) {
131 throw std::out_of_range("Face coordinate is outside the face array");
132 }
133 result += static_cast<std::size_t>(faceCoordinates[axis]) * stride;
134 stride *= static_cast<std::size_t>(dimensions[axis]);
135 }
136 return result;
137}

150PhysicalCoordinates MeshLayout::cellCenter(PhysicalCoordinates const& lower, units::Length cellWidth, Coordinates const& interiorCoordinates) const {
151 PhysicalCoordinates result{};
152 for (int axis = 0; axis < ndim; ++axis) {
153 if (interiorCoordinates[axis] < 0 || interiorCoordinates[axis] >= interiorExtents_[axis]) {
154 throw std::out_of_range("Cell-center coordinate is outside the interior");
155 }
156 result[axis] = lower[axis] + (interiorCoordinates[axis] + Real(0.5)) * cellWidth;
157 }

161void MeshLayout::rebuild() {
162 if (cellsPerActiveDimension_ < 2 || cellsPerActiveDimension_ % 2 != 0) {
163 throw std::invalid_argument("Active mesh extent must be a positive even number");
164 }
165 if (ghostWidth_ < 0) {
166 throw std::invalid_argument("Mesh ghost width cannot be negative");
167 }
168 for (int axis = 0; axis < ndim; ++axis) {
169 interiorExtents_[axis] = cellsPerActiveDimension_;
170 extents_[axis] = cellsPerActiveDimension_ + 2 * ghostWidth_;
171 }
172 strides_[0] = 1;
173 for (int axis = 1; axis < ndim; ++axis) {
174 strides_[axis] = strides_[axis - 1] * extents_[axis - 1];
175 }
176}

196BlockLocation BlockLocation::parent() const {
197 if (isRoot()) {
198 throw std::logic_error("The root block has no parent");
199 }
200 if (level < 0) {
201 throw std::invalid_argument("Invalid block location");
202 }
203 BlockLocation result = *this;

211int BlockLocation::childSlot() const {
212 if (isRoot()) {
213 throw std::logic_error("The root block is not a child");
214 }
215 if (level < 0) {
216 throw std::invalid_argument("Invalid block location");
217 }
218 int result = 0;
219 for (int axis = 0; axis < ndim; ++axis) {
220 result |= (coordinates[axis] & 1) << axis;
221 }

225BlockLocation BlockLocation::child(int slot) const {
226 if (level < 0 || level >= std::numeric_limits<unsigned>::digits - 2 || slot < 0 || slot >= (1 << ndim)) {
227 throw std::invalid_argument("Invalid child slot for block dimensionality");
228 }
229 BlockLocation result = *this;
230 ++result.level;
231 for (int axis = 0; axis < ndim; ++axis) {
232 result.coordinates[axis] = 2 * coordinates[axis] + ((slot >> axis) & 1);
233 }
234 return result;
235}

237std::size_t BlockLocationHash::operator()(BlockLocation const& location) const noexcept {
238 std::size_t result = std::hash<int>{}(location.level);
239 for (int coordinate : location.coordinates) {
240 result ^= std::hash<int>{}(coordinate) + 0x9e3779b9 + (result << 6) + (result >> 2);
241 }
242 return result;
243}
