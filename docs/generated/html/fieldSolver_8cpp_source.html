123 FmmPartition(Config const& config, std::vector<Subgrid> const& blocks, FieldDirectory fields, std::size_t owner, std::size_t partitions)
124 : config_(config)
125 , fields_(std::move(fields))
126 , owner_(owner)
127 , partitions_(partitions) {
128 static_assert(ndim == 3);
129 config_.validate();
130 n_ = config_.mesh.cells * (1 << config_.mesh.level);
131 cellWidth_ = (config_.mesh.upper - config_.mesh.lower) / Real(n_);
132 Real const h = units::value(cellWidth_);
133 int const coefficients = diagonal::coefficientCount(config_.gravity.multipoleOrder);
134 for (int count = 1; count <= n_; count *= 2)
135 levels_.emplace_back(count, h * (n_ / count), coefficients, owner_, partitions_, count == n_);
136#ifdef OCTOTIGERII_WITH_HPX
137 workers_ = hpx::get_os_thread_count();
138 if (config_.runtime.workerTasks > 0) workers_ = std::size_t(config_.runtime.workerTasks);
139#endif
140 workers_ = std::max(std::size_t(1), std::min(workers_, levels_.back().moments.size()));
141 // Each range is contiguous in both the FMM leaf partition and a field
142 // block. This also handles a partition boundary cutting through a block.
143 auto const& leaves = levels_.back();
144 for (std::size_t id = leaves.begin; id < leaves.end; ++id) {
145 auto c = coordinate(id, n_);
146 Coordinate b{};
147 for (int d = 0; d < 3; ++d) {
148 b[d] = c[d] / config_.mesh.cells;
149 c[d] %= config_.mesh.cells;
150 }
151 auto const& block = blocks.at(index(b, 1 << config_.mesh.level));
152 auto const offset = block.interior.offset + block.layout.index(c);
153 if (ranges_.empty() || ranges_.back().range.partition != block.interior.partition ||
154 ranges_.back().range.offset + ranges_.back().range.count != offset)
155 ranges_.push_back({{block.interior.partition, offset, 0}, id - leaves.begin});
156 ++ranges_.back().range.count;

212 template <typename Function>
213 Statistics parallel(std::size_t count, Function const& function) const {
214 if (count == 0) return {};
215 constexpr std::size_t grain = 16;
216 auto const workers = std::min(workers_, (count + grain - 1) / grain);
217 std::atomic<std::size_t> next{0};
218 auto work = [&](std::size_t worker) {
219 Statistics result;
220 result.workerTasks = 1;
221 for (;;) {
222 auto const begin = next.fetch_add(grain, std::memory_order_relaxed);
223 if (begin >= count) break;
224 for (auto i = begin; i < std::min(count, begin + grain); ++i)
225 function(i, worker, result);
226 }
227 return result;
228 };
229 Statistics result;
230#ifdef OCTOTIGERII_WITH_HPX
231 std::vector<hpx::future<Statistics>> pending;
232 pending.reserve(workers);
233 try {
234 for (std::size_t i = 0; i < workers; ++i)
235 pending.push_back(hpx::async([&, i] { return work(i); }));
236 } catch (...) {
237 auto error = std::current_exception();
238 try {
239 collect(pending);
240 } catch (...) {
241 }
242 std::rethrow_exception(error);
243 }
244 for (auto const& part : collect(pending))
245 accumulate(result, part);
246#else
247 (void) workers;
248 result = work(0);
249#endif
250 return result;
251 }

253 // A source belongs to the interaction list if its parent was not accepted.
254 // With theta < 1/sqrt(3), an unaccepted parent implies all its ancestors
255 // were unaccepted as well. Leaves perform the remaining direct interactions.
256 template <typename Function>
257 void interactions(unsigned depth, std::size_t target, Function const& function) const {
258 using std::ceil;
259 using std::hypot;
260
261 auto const& level = levels_[depth];
262 auto const a = coordinate(target, level.count);
263 auto const p = parent(a);
264 int const parents = level.count / 2;
265 Real const theta = config_.gravity.openingAngle;
266 int const radius = theta <= 1 / Real(parents) ? parents : int(ceil(1 / theta));
267 bool const leaf = depth + 1 == levels_.size();
268 for (int z = std::max(0, p[2] - radius); z <= std::min(parents - 1, p[2] + radius); ++z)
269 for (int y = std::max(0, p[1] - radius); y <= std::min(parents - 1, p[1] + radius); ++y)
270 for (int x = std::max(0, p[0] - radius); x <= std::min(parents - 1, p[0] + radius); ++x) {
271 if (hypot(Real(p[0] - x), Real(p[1] - y), Real(p[2] - z)) * theta > 1) continue;
272 for (int slot = 0; slot < 8; ++slot) {
273 auto const b = child({x, y, z}, slot);
274 auto const source = index(b, level.count);
275 if (target == source) continue;
276 diagonal::Offset const r{a[0] - b[0], a[1] - b[1], a[2] - b[2]};
277 if (leaf || hypot(Real(r[0]), Real(r[1]), Real(r[2])) * theta > 1) function(source, r);
278 }
279 }
280 }

282 // Plan on worker threads, deduplicate by owner, and launch all remote reads
283 // before waiting. Only requested source cells are retained in the halo.
284 template <typename Needs>
285 Sources fetch(
286 unsigned depth, bool local, std::size_t targets, Needs const& needs, std::vector<storage::Locality> const& peers, Statistics& statistics) const {
287 auto const& level = levels_.at(depth);
288 Sources sources{level, local, {}};
289 if (partitions_ == 1) return sources;
290 std::vector<std::unordered_set<std::size_t>> requests(workers_);
291 accumulate(statistics, parallel(targets, [&](std::size_t i, std::size_t worker, Statistics&) {
292 needs(i, [&](std::size_t id) {
293 if (id < level.begin || id >= level.end) requests[worker].insert(id);
294 });
295 }));
296 std::unordered_set<std::size_t> unique;
297 for (auto& request : requests)
298 unique.merge(request);
299 std::vector<std::vector<std::size_t>> ids(partitions_);
300 for (auto id : unique)
301 ids.at(id * partitions_ / cellCount(level.count)).push_back(id);
302 sources.remote.reserve(unique.size());
303#ifdef OCTOTIGERII_WITH_HPX
304 // Bound individual parcels, including the coefficient payload.
305 std::size_t const batchSize = std::max(std::size_t(1), std::size_t(65536) / diagonal::coefficientCount(config_.gravity.multipoleOrder));
306 std::vector<std::vector<std::size_t>> batches;
307 std::vector<hpx::future<std::vector<Coefficients>>> pending;
308 try {
309 for (std::size_t owner = 0; owner < partitions_; ++owner) {
310 std::sort(ids[owner].begin(), ids[owner].end());
311 for (std::size_t begin = 0; begin < ids[owner].size(); begin += batchSize) {
312 batches.emplace_back(ids[owner].begin() + begin, ids[owner].begin() + std::min(ids[owner].size(), begin + batchSize));
313 pending.push_back(hpx::async<ReadAction>(peers.at(owner), depth, local, batches.back()));
314 }
315 }
316 } catch (...) {
317 auto error = std::current_exception();
318 try {
319 collect(pending);
320 } catch (...) {
321 }
322 std::rethrow_exception(error);
323 }
324 auto values = collect(pending);
325 for (std::size_t b = 0; b < batches.size(); ++b) {
326 if (values[b].size() != batches[b].size()) throw std::logic_error("Incomplete FMM source transfer");
327 for (std::size_t i = 0; i < batches[b].size(); ++i)
328 sources.remote.emplace(batches[b][i], std::move(values[b][i]));
329 }
330#else
331 (void) peers;
332#endif
333 return sources;
334 }

336 Statistics initialize(unsigned bank) {
337 Statistics result;
338 for (auto& level : levels_)
339 accumulate(result, parallel(level.moments.size(), [&](std::size_t i, std::size_t, Statistics&) {
340 std::fill(level.moments[i].begin(), level.moments[i].end(), 0);
341 std::fill(level.locals[i].begin(), level.locals[i].end(), 0);
342 }));
343 auto const volume = cellWidth_ * cellWidth_ * cellWidth_;
344 auto const gram = units::Mass::from_value(1);
345 accumulate(result, parallel(ranges_.size(), [&](std::size_t r, std::size_t, Statistics&) {
346 auto const& segment = ranges_[r];
347 auto fill = [&](auto density) {
348 for (std::size_t i = 0; i < segment.range.count; ++i) {
349 auto const rho = density(i);
350 if (!(rho >= units::Density{}) || !units::finite(rho) || !units::finite(rho * volume))
351 throw std::invalid_argument("Gravity density/mass must be finite and nonnegative");
352 levels_.back().moments[segment.leaf + i][0] = rho * volume / gram;
353 }
354 };
355 if constexpr (build::hydro) {
356 auto input = std::get<0>(fields_.hydro.fields).read(segment.range, bank).get();
357 fill([&](std::size_t i) { return input.data()[i]; });
358 } else {
359 auto input = fields_.density.read(segment.range, bank).get();
360 fill([&](std::size_t i) { return input.data()[i]; });
361 }
362 }));
363 return result;
364 }

366 Statistics upward(unsigned depth, std::vector<storage::Locality> const& peers) {
367 auto& level = levels_.at(depth);
368 auto const& fine = levels_.at(depth + 1);
369 Statistics result;
370 auto children = fetch(
371 depth + 1, false, level.moments.size(),
372 [&](std::size_t i, auto need) {
373 auto const c = coordinate(level.begin + i, level.count);
374 for (int slot = 0; slot < 8; ++slot)
375 need(index(child(c, slot), fine.count));
376 },
377 peers, result);
378 accumulate(result, parallel(level.moments.size(), [&](std::size_t i, std::size_t, Statistics&) {
379 auto const c = coordinate(level.begin + i, level.count);
380 for (int slot = 0; slot < 8; ++slot)
381 add(level.moments[i],
382 diagonal::shiftMultipole(children.at(index(child(c, slot), fine.count)), childOffset(slot), 0.5, config_.gravity.multipoleOrder));
383 }));
384 return result;
385 }
386
387 Statistics downward(unsigned depth, std::vector<storage::Locality> const& peers) {
388 auto& level = levels_.at(depth);
389 Statistics result;
390 auto sources = fetch(
391 depth, false, level.locals.size(),
392 [&](std::size_t i, auto need) { interactions(depth, level.begin + i, [&](std::size_t id, auto const&) { need(id); }); }, peers, result);
393 auto parents = fetch(
394 depth - 1, true, level.locals.size(),
395 [&](std::size_t i, auto need) { need(index(parent(coordinate(level.begin + i, level.count)), level.count / 2)); }, peers, result);
396 bool const leaf = depth + 1 == levels_.size();
397 accumulate(result, parallel(level.locals.size(), [&](std::size_t i, std::size_t, Statistics& statistics) {
398 auto const target = level.begin + i;
399 interactions(depth, target, [&](std::size_t source, diagonal::Offset const& r) {
400 if (leaf) {
401 diagonal::addDirect(level.locals[i], sources.at(source)[0], r, level.width);
402 if (target < source) ++statistics.directPairs;
403 } else {
404 diagonal::getOperator(config_.gravity.multipoleOrder, r)->add(level.locals[i], sources.at(source), level.width);
405 if (target < source) ++statistics.multipolePairs;
406 }
407 });
408 auto const c = coordinate(target, level.count);
409 int const slot = (c[0] & 1) + 2 * (c[1] & 1) + 4 * (c[2] & 1);
410 add(level.locals[i], diagonal::shiftLocal(parents.at(index(parent(c), level.count / 2)), childOffset(slot), 0.5, config_.gravity.multipoleOrder));
411 }));
412 return result;
413 }

415 template <typename StateType>
416 void copy(storage::ColumnHandle<StateType> const& field, storage::Range range, unsigned bank) const {
417 auto input = field.read(range, bank).get();
418 auto output = field.output(range, bank ^ 1);
419 for (std::size_t i = 0; i < range.count; ++i)
420 output.put(i, input.at(i));
421 field.commit(range, bank ^ 1, output);
422 }
423
424 Statistics publish(unsigned bank) {
425 auto const gram = units::Mass::from_value(1);
426 auto const cm = units::Length::from_value(1);
427 auto result = parallel(ranges_.size(), [&](std::size_t r, std::size_t, Statistics&) {
428 auto const& segment = ranges_[r];
429 auto const range = segment.range;
430 auto output = fields_.gravity.output(range, bank ^ 1);
431 for (std::size_t i = 0; i < range.count; ++i) {
432 auto const& local = levels_.back().locals[segment.leaf + i];
433 State field;
434 field.potential() = constants::G * (gram / cm) * local[0];
435 field.acceleration(0) = -constants::G * (gram / cm) * diagonal::derivative(local, 1, 0, 0) / cellWidth_;
436 field.acceleration(1) = -constants::G * (gram / cm) * diagonal::derivative(local, 0, 1, 0) / cellWidth_;
437 field.acceleration(2) = -constants::G * (gram / cm) * diagonal::derivative(local, 0, 0, 1) / cellWidth_;
438 if (!finite(field)) throw std::runtime_error("Nonfinite gravity solution");
439 output.put(i, field);
440 }
441 fields_.gravity.commit(range, bank ^ 1, output);
442 if constexpr (build::hydro)
443 copy(fields_.hydro, range, bank);
444 else {
445 auto input = fields_.density.read(range, bank).get();
446 auto density = fields_.density.output(range, bank ^ 1);
447 std::copy_n(input.data(), range.count, density.data());
448 fields_.density.commit(range, bank ^ 1, density);
449 }
450 if constexpr (build::radiation) copy(fields_.radiation, range, bank);
451 });
452 result.localityCells = {levels_.back().moments.size()};
453 return result;
454 }

538Statistics FieldSolver::solve(unsigned bank) {
539 auto result = impl_->phase(Stage::Initialize, 0, bank);
540 for (unsigned depth = impl_->depths - 1; depth > 0; --depth)
541 accumulate(result, impl_->phase(Stage::Upward, depth - 1, bank));
542 for (unsigned depth = 1; depth < impl_->depths; ++depth)
543 accumulate(result, impl_->phase(Stage::Downward, depth, bank));
544 auto publication = impl_->phase(Stage::Publish, 0, bank);
545 accumulate(result, publication);
