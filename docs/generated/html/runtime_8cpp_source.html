31std::vector<T> collect(std::vector<hpx::future<T>>& pending) {
32 std::exception_ptr error;
33 std::vector<T> results;
34 for (auto& future : pending) {
35 try {
36 results.push_back(future.get());
37 } catch (...) {
38 if (!error) error = std::current_exception();
39 }
40 }
41 if (error) std::rethrow_exception(error);
42 return results;
43}

203 PhaseResult run(Operation operation, units::Time dt, std::uint64_t generation, std::vector<storage::Locality> const& executors) {
204 if (generation != generation_) throw std::logic_error("Stale stage dispatch");
205#ifdef OCTOTIGERII_WITH_HPX
206 std::vector<hpx::future<PhaseResult>> pending;
207 for (std::size_t i = 0; i < workspaces_.size(); ++i)
208 pending.push_back(hpx::async([&, i] { return worker(operation, dt, generation, executors, workspaces_[i]); }));
209 auto results = collect(pending);
210#else
211 std::vector<PhaseResult> results{worker(operation, dt, generation, executors, workspaces_.front())};
212#endif
213 PhaseResult result;
214 for (auto const& part : results) {
215 result.tasks.localTasks += part.tasks.localTasks;
216 result.tasks.stolenTasks += part.tasks.stolenTasks;
217 result.timestep = std::min(result.timestep, part.timestep);
218 }

280 PhaseResult worker(Operation operation, units::Time dt, std::uint64_t generation, std::vector<storage::Locality> const& executors, Workspace& workspace) {
281 PhaseResult result;
282 auto process = [&](std::uint64_t id, bool stolen) {
283 auto const& block = blocks_.at(id);
284 if (operation == Operation::Timestep)
285 result.timestep = std::min(result.timestep, timestep(block));
286 else if (operation == Operation::Advance) {
287 // Owned plans are immutable. Stolen plans have only metadata and
288 // are temporary, keeping persistent topology caches bounded.
289 std::optional<HaloPlan> temporary;
290 if (stolen) temporary = makeHaloPlan(config_, blocks_, id);
291 auto const& plan = stolen ? *temporary : plans_.at(id);
292 if constexpr (build::hydro) workspace.hydro.advance(block, fields_.hydro, plan, hydro::HydroSystem(config_.hydro.gamma), bank_, dt);
293 if constexpr (build::radiation)
294 workspace.radiation.advance(block, fields_.radiation, plan, radiation::RadiationSystem(config_.radiation.lightSpeedRatio * constants::c), bank_, dt);
295 if constexpr (build::gravity) copyFields(fields_.gravity, block.interior, bank_);
296 } else if constexpr (build::gravity && build::hydro) {
297 kick(block, dt);
298 copyFields(fields_.gravity, block.interior, bank_);
299 } else {
300 throw std::logic_error("Gravity kicks are not part of this executable");
301 }
302 if (stolen)
303 ++result.tasks.stolenTasks;
304 else
305 ++result.tasks.localTasks;
306 };
307 for (;;) {
308 auto work = claim(generation);
309 if (work.empty()) break;
310 process(work.front(), false);
311 }
312#ifdef OCTOTIGERII_WITH_HPX
313 if (config_.runtime.workStealing)
314 for (std::size_t distance = 1; distance < executors.size(); ++distance) {
315 auto const donor = (owner_ + distance) % executors.size();
316 for (;;) {
317 auto work = hpx::async<ClaimAction>(executors[donor], generation).get();
318 if (work.empty()) break;
319 process(work.front(), true);
320 }
321 }
322#else
323 (void) executors;
324#endif
325 return result;
326 }

328 units::Time timestep(Subgrid const& block) const {
329 auto result = units::Time::from_value(std::numeric_limits<Real>::infinity());
330 [[maybe_unused]] auto fieldStep = [&](auto const& fields, auto const& system) {
331 auto input = fields.read(block.interior, bank_).get();
332 units::InverseTime rate{};
333 std::array<units::Velocity, ndim> speed{};
334 for (std::size_t i = 0; i < block.interior.count; ++i)
335 for (int d = 0; d < ndim; ++d)
336 speed[d] = std::max(speed[d], system.maximumSignalSpeed(input.at(i), d));
337 for (auto value : speed)
338 rate += value / block.cellWidth;
339 if (!(rate > units::InverseTime{}) || !units::finite(rate)) throw std::runtime_error("No finite positive signal speed");
340 return config_.timestep.cfl / rate;
341 };
342 if constexpr (build::hydro) {
343 result = fieldStep(fields_.hydro, hydro::HydroSystem(config_.hydro.gamma));
344 if constexpr (build::gravity) {
345 auto input = fields_.gravity.read(block.interior, bank_).get();
346 units::Acceleration acceleration{};
347 for (std::size_t i = 0; i < block.interior.count; ++i) {
348 units::Acceleration norm{};
349 for (int d = 0; d < ndim; ++d)
350 norm += units::abs(input.at(i).acceleration(d));
351 acceleration = std::max(acceleration, norm);
352 }
353 if (acceleration > units::Acceleration{}) {
354 auto const b = config_.timestep.cfl / result;
355 auto const a = 0.5 * acceleration / block.cellWidth;
356 result = 2 * config_.timestep.cfl / (b + units::sqrt(b * b + 4.0 * a * config_.timestep.cfl));
357 result = std::min(result, 0.2 * units::sqrt(block.cellWidth / acceleration));
358 }
359 }
360 }
361 if constexpr (build::radiation)
362 result = std::min(result, fieldStep(fields_.radiation, radiation::RadiationSystem(config_.radiation.lightSpeedRatio * constants::c)));
363 return result;
364 }
365
366 void kick(Subgrid const& block, units::Time dt) {
367 auto input = fields_.hydro.read(block.interior, bank_).get();
368 auto gravity = fields_.gravity.read(block.interior, bank_).get();
369 auto output = fields_.hydro.output(block.interior, bank_ ^ 1);
370 for (std::size_t i = 0; i < block.interior.count; ++i) {
371 auto state = input.at(i);
372 units::EnergyDensity work{};
373 for (int d = 0; d < ndim; ++d) {
374 auto const old = state.momentum(d);
375 auto const impulse = dt * state.density() * gravity.at(i).acceleration(d);
376 state.momentum(d) += impulse;
377 work += impulse * (old + 0.5 * impulse) / state.density();
378 }
379 state.totalEnergy() += work;
380 if (!hydro::HydroSystem(config_.hydro.gamma).admissible(state)) throw std::runtime_error("Invalid gravity kick state");
381 output.put(i, state);
382 }
383 fields_.hydro.commit(block.interior, bank_ ^ 1, output);
384 }

575void Runtime::setGravity(std::vector<std::vector<gravity::State>> const& fields) {
576 std::lock_guard guard(impl_->apiMutex);
577 if (!impl_->config.gravityEnabled() || fields.size() != size()) throw std::invalid_argument("Gravity directory size/field mismatch");
578 for (std::size_t b = 0; b < size(); ++b) {
579 if (fields[b].size() != impl_->topology->blocks()[b].interior.count) throw std::invalid_argument("Gravity block size mismatch");
580 for (auto const& value : fields[b])
581 if (!finite(value)) throw std::invalid_argument("Nonfinite gravity assignment");
582 }
583 // This is a synchronized source-field publication, with no live readers.
584 // Fill the other bank first so a failed transfer cannot corrupt published data.
585 auto const& directory = impl_->fields->directory();
586 for (std::size_t b = 0; b < size(); ++b) {
587 auto const range = impl_->topology->blocks()[b].interior;
588 if (impl_->config.hydroEnabled())
589 copyFields(directory.hydro, range, impl_->bank);
590 else {
591 auto input = directory.density.read(range, impl_->bank).get();
592 auto output = directory.density.output(range, impl_->bank ^ 1);
593 std::copy_n(input.data(), input.size(), output.data());
594 directory.density.commit(range, impl_->bank ^ 1, output);
595 }
596 auto output = directory.gravity.output(range, impl_->bank ^ 1);
597 for (std::size_t i = 0; i < range.count; ++i)
598 output.put(i, fields[b][i]);
599 directory.gravity.commit(range, impl_->bank ^ 1, output);
600 }
601 impl_->bank ^= 1;
602 ++impl_->generation;
603 impl_->gravityTime = impl_->time.time;
604 impl_->gravityReady = true;
605}
