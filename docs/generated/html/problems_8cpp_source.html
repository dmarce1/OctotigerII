12void initializeProblem(Snapshot& data, Config const& c) {
13 using std::exp;
14 using std::round;
15 using std::sin;
16 using std::sqrt;
17 using std::tanh;
18
19 hydro::HydroSystem gas(c.hydro.gamma);
20 data.layout.forEachInterior([&](mesh::Coordinates const& cell, std::size_t i) {
21 auto const point = data.layout.cellCenter(data.lower, data.cellWidth, cell);
22 auto const length = c.mesh.upper - c.mesh.lower;
23 Real radius2 = 0;
24 for (int d = 0; d < c.dimensions; ++d) {
25 Real const r = (point[d] - (c.mesh.lower + c.mesh.upper) / 2.0) / length;
26 radius2 += r * r;
27 }
28 Real const gaussian = exp(-radius2 / (2 * 0.15 * 0.15));
29 if (data.hydroEnabled) {
30 hydro::PrimitiveState primitive{};
31 primitive.density() = units::Density::from_value(1);
32 primitive.pressure() = units::Pressure::from_value(1);
33 if (c.problem == "sod") {
34 bool const left = point[0] < (c.mesh.lower + c.mesh.upper) / 2.0;
35 primitive.density() = units::Density::from_value(left ? 1 : 0.125);
36 primitive.pressure() = units::Pressure::from_value(left ? 1 : 0.1);
37 } else if (c.problem == "kelvin-helmholtz") {
38 Real const x = (point[0] - c.mesh.lower) / length, y = (point[1] - c.mesh.lower) / length;
39 Real const layer = 0.5 * (tanh((y - 0.25) / 0.025) - tanh((y - 0.75) / 0.025));
40 primitive.density() = units::Density::from_value(1 + layer);
41 primitive.pressure() = units::Pressure::from_value(2.5);
42 primitive.velocity(0) = units::Velocity::from_value(layer - 0.5);
43 primitive.velocity(1) = units::Velocity::from_value(0.01 * sin(4 * piR * x));
44 } else if (c.problem == "collapse") {
45 primitive.density() = units::Density::from_value(1e4 * (0.01 + gaussian));
46 primitive.pressure() = units::Pressure::from_value(1e12);
47 }
48 data.hydro.values()[i] = gas.conservedState(primitive);
49 }
50 if (data.radiationEnabled) {
51 Real distance2 = 0;
52 for (int axis = 0; axis < c.dimensions; ++axis) {
53 auto distance = point[axis] - (c.mesh.lower + 0.25 * length);
54 if (c.mesh.periodic) distance -= round(distance / length) * length;
55 distance2 += distance * distance / (0.08 * length * 0.08 * length);
56 }
57 auto& state = data.radiation.values()[i];
58 state.energy() = units::EnergyDensity::from_value(1e-6 + exp(-0.5 * distance2));
59 if (c.problem == "streaming")
60 for (int axis = 0; axis < c.dimensions; ++axis)
61 state.radiativeFlux(axis) = constants::c * state.energy() / sqrt(Real(c.dimensions));
62 }
63 if (data.gravityEnabled && !data.hydroEnabled)
64 data.density.values()[i] = units::Density::from_value(1e4 * (c.problem == "gravity-sphere" ? Real(radius2 < 0.25 * 0.25) : gaussian));
65 });
66}
