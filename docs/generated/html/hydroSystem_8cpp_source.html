27PrimitiveState HydroSystem::reconstructionVariables(ConservedState const& state) const {
28 if (!admissible(state)) {
29 throw std::runtime_error("Cannot convert an inadmissible hydro state to primitive variables");
30 }
31 PrimitiveState result;
32 result.setDensity(state.density());
33 for (int axis = 0; axis < ndim; ++axis) {
34 result.setVelocity(axis, state.momentum(axis) / state.density());
35 }
36 result.setPressure(pressure(state));
37 return result;
38}

40ConservedState HydroSystem::conservedState(PrimitiveState const& state) const {
41 if (!(state.density() >= densityFloor_) || !(state.pressure() >= pressureFloor_) || !units::finite(state.density()) || !units::finite(state.pressure())) {
42 throw std::invalid_argument("Cannot convert an inadmissible hydro primitive state");
43 }
44 ConservedState result;
45 result.setDensity(state.density());
46 units::VelocitySquared speedSquared{};
47 for (int axis = 0; axis < ndim; ++axis) {
48 if (!units::finite(state.velocity(axis))) {
49 throw std::invalid_argument("Hydro primitive velocity must be finite");
50 }
51 result.setMomentum(axis, state.density() * state.velocity(axis));
52 speedSquared += state.velocity(axis) * state.velocity(axis);
53 }
54 result.setTotalEnergy(state.pressure() / (adiabaticIndex_ - 1) + Real(0.5) * state.density() * speedSquared);
55 return result;
56}

58ConservedFlux HydroSystem::physicalFlux(ConservedState const& state, int normal) const {
59 PrimitiveState const primitive = reconstructionVariables(state);
60 auto const normalVelocity = primitive.velocity(normal);
61 ConservedFlux result;
62 result.setMass(state.density() * normalVelocity);
63 for (int axis = 0; axis < ndim; ++axis) {
64 result.setMomentum(axis, state.momentum(axis) * normalVelocity + (axis == normal ? primitive.pressure() : units::Pressure{}));
65 }
66 result.setEnergy((state.totalEnergy() + primitive.pressure()) * normalVelocity);
67 return result;
68}

70ConservedFlux HydroSystem::riemann(ConservedState const& left, ConservedState const& right, int normal) const {
71 PrimitiveState const leftPrimitive = reconstructionVariables(left);
72 PrimitiveState const rightPrimitive = reconstructionVariables(right);
73 auto const leftSound = units::sqrt(adiabaticIndex_ * leftPrimitive.pressure() / leftPrimitive.density());
74 auto const rightSound = units::sqrt(adiabaticIndex_ * rightPrimitive.pressure() / rightPrimitive.density());
75 auto const leftSpeed = std::min(leftPrimitive.velocity(normal) - leftSound, rightPrimitive.velocity(normal) - rightSound);
76 auto const rightSpeed = std::max(leftPrimitive.velocity(normal) + leftSound, rightPrimitive.velocity(normal) + rightSound);
77 ConservedFlux const leftFlux = physicalFlux(left, normal);
78 ConservedFlux const rightFlux = physicalFlux(right, normal);
79 if (leftSpeed >= units::Velocity{}) {
80 return leftFlux;
81 }
82 if (rightSpeed <= units::Velocity{}) {
83 return rightFlux;
84 }
85
86 auto const leftDenominator = leftPrimitive.density() * (leftSpeed - leftPrimitive.velocity(normal));
87 auto const rightDenominator = rightPrimitive.density() * (rightSpeed - rightPrimitive.velocity(normal));
88 auto const denominator = leftDenominator - rightDenominator;
89 auto const denominatorScale = std::max({units::MassFlux::from_value(1), units::abs(leftDenominator), units::abs(rightDenominator)});
90 auto const waveScale = std::max({units::Velocity::from_value(1), units::abs(leftSpeed), units::abs(rightSpeed)});
91 if (units::abs(denominator) <= 64 * epsilonR * denominatorScale) {
92 return hll(left, right, normal);
93 }
94 auto const contactSpeed = (rightPrimitive.pressure() - leftPrimitive.pressure() +
95 leftPrimitive.density() * leftPrimitive.velocity(normal) * (leftSpeed - leftPrimitive.velocity(normal)) -
96 rightPrimitive.density() * rightPrimitive.velocity(normal) * (rightSpeed - rightPrimitive.velocity(normal))) /
97 denominator;
98 if (!units::finite(contactSpeed) || contactSpeed < leftSpeed || contactSpeed > rightSpeed) {
99 return hll(left, right, normal);
100 }
101
102 auto starState = [&](ConservedState const& state, PrimitiveState const& primitive, units::Velocity waveSpeed) {
103 auto const waveDifference = waveSpeed - primitive.velocity(normal);
104 auto const starDifference = waveSpeed - contactSpeed;
105 if (units::abs(starDifference) <= 64 * epsilonR * waveScale || units::abs(waveDifference) <= 64 * epsilonR * waveScale) {
106 return ConservedState{};
107 }
108 ConservedState star;
109 star.setDensity(primitive.density() * waveDifference / starDifference);
110 for (int axis = 0; axis < ndim; ++axis) {
111 auto const velocity = axis == normal ? contactSpeed : primitive.velocity(axis);
112 star.setMomentum(axis, star.density() * velocity);
113 }
114 star.setTotalEnergy(star.density() *
115 (state.totalEnergy() / primitive.density() +
116 (contactSpeed - primitive.velocity(normal)) * (contactSpeed + primitive.pressure() / (primitive.density() * waveDifference))));
117 return star;
118 };
119
120 if (contactSpeed >= units::Velocity{}) {
121 ConservedState const star = starState(left, leftPrimitive, leftSpeed);
122 if (!admissible(star)) {
123 return hll(left, right, normal);
124 }
125 return leftFlux + advectiveFlux(star - left, leftSpeed);
126 }
127 ConservedState const star = starState(right, rightPrimitive, rightSpeed);
128 if (!admissible(star)) {
129 return hll(left, right, normal);
130 }
131 return rightFlux + advectiveFlux(star - right, rightSpeed);
132}

154ConservedFlux HydroSystem::limitFlux(
155 ConservedState const& left, ConservedState const& right, ConservedFlux const& highOrderFlux, int normal, units::TimePerLength stepOverCellWidth) const {
156 if (stepOverCellWidth == units::TimePerLength{}) {
157 return highOrderFlux;
158 }
159 ConservedFlux const leftPhysical = physicalFlux(left, normal);
160 ConservedFlux const rightPhysical = physicalFlux(right, normal);
161 auto const factor = Real(2 * ndim) * stepOverCellWidth;
162 auto validFlux = [&](ConservedFlux const& flux) {
163 return admissible(left - integratedFlux(flux - leftPhysical, factor)) && admissible(right + integratedFlux(flux - rightPhysical, factor));
164 };
165 if (validFlux(highOrderFlux)) {
166 return highOrderFlux;
167 }
168 auto const speed = std::max(maximumSignalSpeed(left, normal), maximumSignalSpeed(right, normal));
169 ConservedFlux const lowOrderFlux = Real(0.5) * (leftPhysical + rightPhysical - advectiveFlux(right - left, speed));
170 if (!validFlux(lowOrderFlux)) {
171 throw std::runtime_error("First-order hydro flux is not positivity preserving at this timestep");
172 }
173 Real low = 0;
174 Real high = 1;
175 for (int iteration = 0; iteration < 56; ++iteration) {
176 Real const fraction = Real(0.5) * (low + high);
177 if (validFlux(lowOrderFlux + fraction * (highOrderFlux - lowOrderFlux))) {
178 low = fraction;
179 } else {
180 high = fraction;
181 }
182 }
183 return lowOrderFlux + low * (highOrderFlux - lowOrderFlux);
184}

186units::Pressure HydroSystem::pressure(ConservedState const& state) const {
187 if (!(state.density() > units::Density{}) || !units::finite(state.density())) {
188 return units::Pressure::from_value(-std::numeric_limits<Real>::infinity());
189 }
190 decltype(units::MomentumDensity{} * units::MomentumDensity{}) momentumSquared{};
191 for (int axis = 0; axis < ndim; ++axis) {
192 momentumSquared += state.momentum(axis) * state.momentum(axis);
193 }
194 return (adiabaticIndex_ - 1) * (state.totalEnergy() - Real(0.5) * momentumSquared / state.density());
195}

197ConservedFlux HydroSystem::hll(ConservedState const& left, ConservedState const& right, int normal) const {
198 PrimitiveState const leftPrimitive = reconstructionVariables(left);
199 PrimitiveState const rightPrimitive = reconstructionVariables(right);
200 auto const leftSound = units::sqrt(adiabaticIndex_ * leftPrimitive.pressure() / leftPrimitive.density());
201 auto const rightSound = units::sqrt(adiabaticIndex_ * rightPrimitive.pressure() / rightPrimitive.density());
202 auto const leftSpeed = std::min(leftPrimitive.velocity(normal) - leftSound, rightPrimitive.velocity(normal) - rightSound);
203 auto const rightSpeed = std::max(leftPrimitive.velocity(normal) + leftSound, rightPrimitive.velocity(normal) + rightSound);
204 ConservedFlux const leftFlux = physicalFlux(left, normal);
205 ConservedFlux const rightFlux = physicalFlux(right, normal);
206 if (leftSpeed >= units::Velocity{}) {
207 return leftFlux;
208 }
209 if (rightSpeed <= units::Velocity{}) {
210 return rightFlux;
211 }
212 return Real(rightSpeed / (rightSpeed - leftSpeed)) * leftFlux - Real(leftSpeed / (rightSpeed - leftSpeed)) * rightFlux +
213 advectiveFlux(right - left, leftSpeed * rightSpeed / (rightSpeed - leftSpeed));
214}

216ConservedFlux HydroSystem::advectiveFlux(ConservedState const& state, units::Velocity speed) {
217 ConservedFlux result;
218 result.setMass(state.density() * speed);
219 for (int axis = 0; axis < ndim; ++axis)
220 result.setMomentum(axis, state.momentum(axis) * speed);
221 result.setEnergy(state.totalEnergy() * speed);
222 return result;
223}
