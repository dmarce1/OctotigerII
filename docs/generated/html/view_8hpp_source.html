66template <typename State>
67void readHalo(storage::ColumnHandle<State> const& fields, HaloPlan const& plan, unsigned bank, std::vector<State>& ghosts) {
68 std::vector<storage::PendingColumns<State>> pending;
69 pending.reserve(plan.reads.size());
70 // Launch all independent reads before awaiting any result.
71 for (auto const& read : plan.reads)
72 pending.push_back(fields.read(read.range, bank));
73 ghosts.resize(plan.ghostCount);
74 std::exception_ptr error;
75 for (std::size_t i = 0; i < pending.size(); ++i) {
76 try {
77 auto values = pending[i].get();
78 for (auto const& copy : plan.reads[i].copies)
79 ghosts[copy.destination] = values.at(copy.source);
80 } catch (...) {
81 if (!error) error = std::current_exception();
82 }
83 }
84 if (error) std::rethrow_exception(error);
85}
