34inline std::size_t linearIndex(Coordinates const& cell, Coordinates const& extents) {
35 std::size_t index = 0, stride = 1;
36 for (int axis = 0; axis < ndim; ++axis) {
37 if (cell[axis] < 0 || cell[axis] >= extents[axis]) throw std::out_of_range("Coordinate outside extent");
38 index += static_cast<std::size_t>(cell[axis]) * stride;
39 stride *= extents[axis];
40 }
41 return index;
42}

46template <typename Function>
47void forEachCoordinate(Coordinates const& begin, Coordinates const& end, Function&& function) {
48 for (int axis = 0; axis < ndim; ++axis)
49 if (end[axis] <= begin[axis]) return;
50 Coordinates cell = begin;
51 while (true) {
52 function(cell);
53 int axis = 0;
54 for (; axis < ndim; ++axis) {
55 if (++cell[axis] < end[axis]) break;
56 cell[axis] = begin[axis];
57 }
58 if (axis == ndim) return;
59 }
60}

62template <typename Function>
63void forEachCoordinate(Coordinates const& extents, Function&& function) {
64 forEachCoordinate(Coordinates{}, extents, std::forward<Function>(function));
65}
