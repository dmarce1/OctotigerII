51class Layout {
52public:

55
56 Layout(std::vector<std::size_t> const& counts, std::size_t partitions) {
57 if (partitions == 0) throw std::invalid_argument("Storage needs at least one partition");
58 std::vector<std::size_t> sizes(partitions, 0);
59 for (std::size_t i = 0; i < counts.size(); ++i) {
60 if (!counts[i]) throw std::invalid_argument("Empty record allocation");
61 // Contiguous chunks preserve the order chosen by the topology adapter.
62 auto const owner = std::min(partitions - 1, i * partitions / counts.size());
63 if (counts[i] > std::numeric_limits<std::size_t>::max() - sizes[owner]) throw std::overflow_error("Field allocation overflow");
64 ranges_.push_back({owner, sizes[owner], counts[i]});
65 sizes[owner] += counts[i];
66 }
67 capacities_ = std::move(sizes);
68 }
69

71 return ranges_;
72 }
