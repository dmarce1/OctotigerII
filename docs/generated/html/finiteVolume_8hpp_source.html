64template <typename Q>
65Q limitedSlope(Q leftDifference, Q rightDifference, Limiter limiter, Real theta = Real(1.5)) {
66 if (!((leftDifference > Q{} && rightDifference > Q{}) || (leftDifference < Q{} && rightDifference < Q{}))) {
67 return Q{};
68 }
69 Real const sign = (leftDifference > Q{} ? Real(1) : Real(-1));
70 auto const left = boost::units::abs(leftDifference);
71 auto const right = boost::units::abs(rightDifference);
72 switch (limiter) {
73 case Limiter::Minmod:
74 return sign * std::min(left, right);
75 case Limiter::VanLeer:
76 return sign * (Real(2) * left * right / (left + right));
77 case Limiter::MinmodTheta: {
78 auto const centered = Real(0.5) * (left + right);
79 return sign * std::min({theta * left, centered, theta * right});
80 }
81 }
82 throw std::logic_error("Unknown finite-volume limiter");
83}

87template <typename System>
88void fillGhostCells(mesh::PatchData<typename System::State>& patch, BoundaryConditions const& boundaries, System const& system) {
89 using State = typename System::State;
90 mesh::MeshLayout const& layout = patch.layout();
91 int const ghostWidth = layout.ghostWidth();
92 if (ghostWidth == 0) {
93 return;
94 }
95 mesh::Coordinates const extents = layout.extents();
96 mesh::forEachCoordinate(extents, [&](mesh::Coordinates const& destination) {
97 if (layout.isInterior(destination)) {
98 return;
99 }
100 mesh::Coordinates source = destination;
101 std::array<bool, ndim> reflect{};
102 for (int axis = 0; axis < ndim; ++axis) {
103 int coordinate = destination[axis] - ghostWidth;
104 int const count = layout.cellsPerActiveDimension();
105 if (coordinate < 0) {
106 switch (boundaries.lower[axis]) {
107 case BoundaryCondition::Periodic:
108 coordinate = (coordinate % count + count) % count;
109 break;
110 case BoundaryCondition::Outflow:
111 coordinate = 0;
112 break;
113 case BoundaryCondition::Reflecting:
114 coordinate = -coordinate - 1;
115 reflect[axis] = true;
116 break;
117 }
118 } else if (coordinate >= count) {
119 switch (boundaries.upper[axis]) {
120 case BoundaryCondition::Periodic:
121 coordinate %= count;
122 break;
123 case BoundaryCondition::Outflow:
124 coordinate = count - 1;
125 break;
126 case BoundaryCondition::Reflecting:
127 coordinate = 2 * count - coordinate - 1;
128 reflect[axis] = true;
129 break;
130 }
131 }
132 source[axis] = coordinate + ghostWidth;
133 }
134 State state = patch.atStorage(source);
135 for (int axis = 0; axis < ndim; ++axis) {
136 if (reflect[axis]) {
137 state = system.reflected(state, axis);
138 }
139 }
140 patch.atStorage(destination) = state;
141 });
142}

184 template <typename Patch>
185 units::Time stableTimestep(Patch const& patch, Real courantNumber) const {
186 if (!(courantNumber > 0 && courantNumber <= Real(0.5))) {
187 throw std::invalid_argument("Invalid MUSCL-Hancock layout or Courant number");
188 }
189 std::array<units::Velocity, ndim> maximumSpeed{};
190 patch.layout().forEachInterior([&](mesh::Coordinates const& cell, std::size_t) {
191 State const state = patch.atInterior(cell);
192 for (int axis = 0; axis < ndim; ++axis) {
193 maximumSpeed[axis] = std::max(maximumSpeed[axis], system_.maximumSignalSpeed(state, axis));
194 }
195 });
196 units::InverseTime inverseStep{};
197 for (auto speed : maximumSpeed) {
198 inverseStep += speed / patch.cellWidth();
199 }
200 if (!(inverseStep > units::InverseTime{}) || !units::finite(inverseStep)) {
201 throw std::runtime_error("No finite positive signal speed in finite-volume patch");
202 }
203 return courantNumber / inverseStep;
204 }

212 // AMR drivers can supply same-level, coarse/fine, or physical boundary
213 // data at the requested physical time. The time-tagged interface avoids
214 // baking a global-cycle assumption into the spatial integrator and can
215 // later interpolate boundaries for level-dependent timesteps.
217 template <typename BoundaryUpdater>
218 StepResult advanceWithBoundaryUpdater(mesh::PatchData<State>& patch, units::Time stepSize, BoundaryUpdater&& updateBoundaries) const {
219 mesh::TimeInterval const interval{patch.timeState().time, patch.timeState().time + stepSize};
220 updateBoundaries(patch, interval.begin);
221 Workspace workspace;
222 auto next = patch.values();
223 advanceInto(patch, stepSize, workspace,
224 [&](mesh::Coordinates const& cell, State const& value) { next[patch.layout().index(patch.layout().storageCoordinates(cell))] = value; });
225 patch.values().swap(next);
226 patch.timeState().completeStep(stepSize);
227 updateBoundaries(patch, interval.end);
228 return StepResult{interval, std::move(workspace.fluxes)};
229 }

242 // Input is immutable. The caller owns a disjoint output range and publishes
243 // it only after every task in the stage succeeds. Workspace is reusable.
247 template <typename Patch, typename Writer>
248 void advanceInto(Patch const& patch, units::Time stepSize, Workspace& workspace, Writer&& write) const {
249 mesh::MeshLayout const& layout = patch.layout();
250 if (layout.ghostWidth() < 2) throw std::invalid_argument("MUSCL-Hancock needs two ghost cells");
251 if (!(stepSize > units::Time{}) || !units::finite(stepSize)) throw std::invalid_argument("MUSCL-Hancock timestep must be positive and finite");
252 auto& minus = workspace.minus;
253 auto& plus = workspace.plus;
254 for (int axis = 0; axis < ndim; ++axis) {
255 minus[axis].resize(layout.cellCount());
256 plus[axis].resize(layout.cellCount());
257 }
258 predictFaceStates(patch, stepSize, minus, plus);
259
260 auto& fluxes = workspace.fluxes;
261 for (int axis = 0; axis < ndim; ++axis) {
262 fluxes[axis].resize(layout.faceCount(axis));
263 mesh::Coordinates const faceExtents = layout.faceExtents(axis);
264 mesh::forEachCoordinate(faceExtents, [&](mesh::Coordinates const& face) {
265 mesh::Coordinates left = face;
266 for (int d = 0; d < ndim; ++d)
267 left[d] = std::min(left[d], layout.interiorExtent(d) - 1);
268 left = layout.storageCoordinates(left);
269 mesh::Coordinates right = left;
270 left[axis] = layout.ghostWidth() + face[axis] - 1;
271 right[axis] = layout.ghostWidth() + face[axis];
272 State const& leftState = plus[axis][layout.index(left)];
273 State const& rightState = minus[axis][layout.index(right)];
274 Flux highOrderFlux = system_.riemann(leftState, rightState, axis);
275 highOrderFlux = system_.limitFlux(patch.atStorage(left), patch.atStorage(right), highOrderFlux, axis, stepSize / patch.cellWidth());
276 fluxes[axis][layout.faceIndex(axis, face)] = highOrderFlux;
277 });
278 }
279
280 layout.forEachInterior([&](mesh::Coordinates const& cell, std::size_t) {
281 Flux update{};
282 for (int axis = 0; axis < ndim; ++axis) {
283 mesh::Coordinates lowerFace = cell;
284 mesh::Coordinates upperFace = cell;
285 ++upperFace[axis];
286 update += fluxes[axis][layout.faceIndex(axis, lowerFace)] - fluxes[axis][layout.faceIndex(axis, upperFace)];
287 }
288 State candidate = patch.atInterior(cell) + (stepSize / patch.cellWidth()) * update;
289 candidate = system_.correctRoundoff(candidate, componentAbs((stepSize / patch.cellWidth()) * update));
290 if (!system_.admissible(candidate)) {
291 throw std::runtime_error("MUSCL-Hancock update produced an inadmissible state");
292 }
293 write(cell, candidate);
294 });
295 }

305 template <typename Patch>
306 void predictFaceStates(
307 Patch const& patch, units::Time stepSize, std::array<std::vector<State>, ndim>& minus, std::array<std::vector<State>, ndim>& plus) const {
308 mesh::MeshLayout const& layout = patch.layout();
309 int const ghostWidth = layout.ghostWidth();
310 auto const begin = mesh::filledCoordinates(ghostWidth - 1);
311 auto const end = mesh::filledCoordinates(ghostWidth + layout.cellsPerActiveDimension() + 1);
312
313 mesh::forEachCoordinate(begin, end, [&](mesh::Coordinates const& cell) {
314 std::size_t const cellIndex = layout.index(cell);
315 State const centerState = patch.atStorage(cell);
316 Reconstruction const center = system_.reconstructionVariables(centerState);
317 std::array<Reconstruction, ndim> slopes{};
318 for (int axis = 0; axis < ndim; ++axis) {
319 mesh::Coordinates left = cell;
320 mesh::Coordinates right = cell;
321 --left[axis];
322 ++right[axis];
323 Reconstruction const leftState = system_.reconstructionVariables(patch.atStorage(left));
324 Reconstruction const rightState = system_.reconstructionVariables(patch.atStorage(right));
325 slopes[axis].forEach([&](auto field, auto& slope) {
326 slope = limitedSlope(center.template get<field>() - leftState.template get<field>(),
327 rightState.template get<field>() - center.template get<field>(), limiter_, theta_);
328 });
329 }
330
331 Real slopeFraction = 1;
332 auto validFaces = [&](Real fraction) {
333 for (int axis = 0; axis < ndim; ++axis) {
334 State const lower = system_.conservedState(center - Real(0.5) * fraction * slopes[axis]);
335 State const upper = system_.conservedState(center + Real(0.5) * fraction * slopes[axis]);
336 if (!system_.admissible(lower) || !system_.admissible(upper)) {
337 return false;
338 }
339 }
340 return true;
341 };
342 if (!validFaces(slopeFraction)) {
343 Real low = 0;
344 Real high = 1;
345 for (int iteration = 0; iteration < 48; ++iteration) {
346 Real const fraction = Real(0.5) * (low + high);
347 if (validFaces(fraction)) {
348 low = fraction;
349 } else {
350 high = fraction;
351 }
352 }
353 slopeFraction = low;
354 }
355
356 Flux predictorFlux{};
357 for (int axis = 0; axis < ndim; ++axis) {
358 minus[axis][cellIndex] = system_.conservedState(center - Real(0.5) * slopeFraction * slopes[axis]);
359 plus[axis][cellIndex] = system_.conservedState(center + Real(0.5) * slopeFraction * slopes[axis]);
360 predictorFlux += system_.physicalFlux(minus[axis][cellIndex], axis) - system_.physicalFlux(plus[axis][cellIndex], axis);
361 }
362 State const predictor = (Real(0.5) * stepSize / patch.cellWidth()) * predictorFlux;
363 bool validPrediction = true;
364 for (int axis = 0; axis < ndim; ++axis) {
365 minus[axis][cellIndex] += predictor;
366 plus[axis][cellIndex] += predictor;
367 validPrediction = validPrediction && system_.admissible(minus[axis][cellIndex]) && system_.admissible(plus[axis][cellIndex]);
368 }
369 if (!validPrediction) {
370 for (int axis = 0; axis < ndim; ++axis) {
371 minus[axis][cellIndex] = centerState;
372 plus[axis][cellIndex] = centerState;
373 }
374 }
375 });
376 }
377};
