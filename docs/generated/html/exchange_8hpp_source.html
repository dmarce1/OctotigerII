46inline void validateDomain(ExchangeDomain const& domain) {
47 if (domain.dimensionCount < 1 || domain.dimensionCount > 3 || !units::finite(domain.lower) ||
48 !units::finite(domain.upper) || !(domain.upper > domain.lower) ||
49 !units::finite(domain.upper - domain.lower))
50 throw std::invalid_argument(
51 "Transport domain must have 1-3 dimensions and finite increasing bounds");
52}

54inline bool sameTime(units::Time left, units::Time right) {
55 return units::finite(left) && units::finite(right) &&
56 units::abs(left - right) <=
57 64 * epsilonR * std::max(units::abs(left), units::abs(right));
58}

60template <class State>
61void fillHalo(mesh::PatchData<State>& target, std::vector<FieldSnapshot<State>> const& sources,
62 ExchangeDomain const& domain, units::Time time, bool includeInterior = false) {
63 validateDomain(domain);
64 auto const& layout = target.layout();
65 if (layout.dimensionCount() != domain.dimensionCount || !(domain.upper > domain.lower) ||
66 !sameTime(target.timeState().time, time))
67 throw std::invalid_argument("Invalid halo domain or target physical time");
68 for (auto const& source : sources) {
69 if (source.fields.layout().dimensionCount() != domain.dimensionCount ||
70 !sameTime(source.fields.timeState().time, time))
71 throw std::invalid_argument(
72 "Transport halo snapshot has wrong dimension or physical time");
73 }
74 auto const extents = layout.extents();
75 auto const width = target.cellWidth();
76 // Accumulate dimensionless fractions of the target cell volume.
77 for (int z = 0; z < extents[2]; ++z)
78 for (int y = 0; y < extents[1]; ++y)
79 for (int x = 0; x < extents[0]; ++x) {
80 mesh::Coordinates const storage{x, y, z};
81 if (!includeInterior && layout.isInterior(storage))
82 continue;
83 auto cell = storage;
84 for (int axis = 0; axis < domain.dimensionCount; ++axis)
85 cell[axis] -= layout.ghostWidth();
86 mesh::PhysicalCoordinates low{};
87 for (int axis = 0; axis < domain.dimensionCount; ++axis) {
88 low[axis] = target.lower()[axis] + Real(cell[axis]) * width;
89 if (domain.periodic) {
90 auto const length = domain.upper - domain.lower;
91 low[axis] -= std::floor((low[axis] - domain.lower) / length) * length;
92 } else {
93 low[axis] = std::clamp(low[axis], domain.lower, domain.upper - width);
94 }
95 }
96 State sum{};
97 Real coverage = 0;
98 for (auto const& source : sources) {
99 auto const& patch = source.fields;
100 auto const& sourceLayout = patch.layout();
101 auto const sourceWidth = patch.cellWidth();
102 mesh::Coordinates first{}, last{};
103 bool overlaps = true;
104 for (int axis = 0; axis < domain.dimensionCount; ++axis) {
105 first[axis] =
106 std::max(0, static_cast<int>(std::floor(
107 (low[axis] - patch.lower()[axis]) / sourceWidth)));
108 last[axis] =
109 std::min(sourceLayout.interiorExtent(axis),
110 static_cast<int>(std::ceil(
111 (low[axis] + width - patch.lower()[axis]) / sourceWidth)));
112 overlaps = overlaps && first[axis] < last[axis];
113 }
114 for (int axis = domain.dimensionCount; axis < 3; ++axis)
115 last[axis] = 1;
116 if (!overlaps)
117 continue;
118 for (int k = first[2]; k < last[2]; ++k)
119 for (int j = first[1]; j < last[1]; ++j)
120 for (int i = first[0]; i < last[0]; ++i) {
121 mesh::Coordinates const donor{i, j, k};
122 Real overlap = 1;
123 for (int axis = 0; axis < domain.dimensionCount; ++axis) {
124 auto const donorLow =
125 patch.lower()[axis] + Real(donor[axis]) * sourceWidth;
126 overlap *= std::max(units::Length{},
127 std::min(low[axis] + width,
128 donorLow + sourceWidth) -
129 std::max(low[axis], donorLow)) /
130 width;
131 }
132 sum += overlap * patch.atInterior(donor);
133 coverage += overlap;
134 }
135 }
136 if (std::abs(coverage - Real(1)) > 256 * epsilonR)
137 throw std::runtime_error(
138 "Halo directory has a gap, overlap, or unsupported periodic cell");
139 target.atStorage(storage) = sum / coverage;
140 }
141}

143template <int dimensions, class State, class System>
144FieldFluxPacket<typename System::Flux>
145advance(mesh::BlockLocation const& location, mesh::PatchData<State>& fields,
146 std::vector<FieldSnapshot<State>> const& sources, ExchangeDomain const& domain,
147 System const& system, units::Time stepSize) {
148 physics::MusclHancock<System, dimensions> solver{system};
149 auto const begin = fields.timeState().time;
150 auto result = solver.advanceWithBoundaryUpdater(
151 fields, stepSize, [&](mesh::PatchData<State>& target, units::Time requestedTime) {
152 // End-state halos require the global end-state barrier. They must
153 // NOT be refreshed using old snapshots mislabeled as new time.
154 if (requestedTime == begin) {
155 fillHalo(target, sources, domain, requestedTime);
156 }
157 });
158 FieldFluxPacket<typename System::Flux> packet{
159 location, fields.layout(), fields.lower(), fields.cellWidth(), result.timeInterval, {}};
160 for (auto& flux : result.faceFluxes)
161 packet.fluxes.push_back(std::move(flux));
162 return packet;
163}
164template <class State, class System>
165units::Time stableStep(mesh::PatchData<State> const& fields, System const& system, Real cfl) {
166 switch (fields.layout().dimensionCount()) {
167 case 1:
168 return physics::MusclHancock<System, 1>(system).stableTimestep(fields, cfl);
169 case 2:
170 return physics::MusclHancock<System, 2>(system).stableTimestep(fields, cfl);
171 case 3:
172 return physics::MusclHancock<System, 3>(system).stableTimestep(fields, cfl);
173 }
174 throw std::logic_error("Invalid transport dimensions");
175}
