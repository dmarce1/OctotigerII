45Reference reference(Config const& c, units::Time time) {
46 Reference result;
47 if (c.verification.analytic == "off") {
48 result.reason = "Analytic comparison disabled";
49 return result;
50 }
51 if (usesDirectGravity(c)) {
52 result.name = "Direct summation of cell-center masses";
53 result.reason = "Direct gravity comparison requires all snapshots; no full-grid exact Silo fields are generated";
54 return result;
55 }
56 result = problemReference(c);
57 if (time > result.validUntil) result.evaluate = {};
58 if (!result.evaluate && (c.verification.analytic == "on" || c.verification.relativeL1Tolerance >= 0))
59 throw std::invalid_argument("Analytic comparison unavailable: " + result.reason);
60 return result;
61}

122Comparison compare(std::vector<Snapshot> const& snapshots, Config const& c) {
123 using std::abs;
124 using std::sqrt;
125
126 if (snapshots.empty()) throw std::invalid_argument("Analytic comparison needs snapshots");
127
128#if OCTOTIGERII_GRAVITY
129 if (usesDirectGravity(c)) return compareDirectGravity(snapshots, c);
130#endif
131 Comparison result;
132 result.time = snapshots.front().time;
133 result.cellsPerAxis = c.mesh.cells * (1 << c.mesh.level);
134 result.multipoleOrder = c.gravity.multipoleOrder;
135 result.openingAngle = c.gravity.openingAngle;
136 auto const ref = reference(c, result.time);
137 result.name = ref.name;
138 result.status = ref.evaluate ? "available" : c.verification.analytic == "off" ? "disabled" : "unavailable";
139 if (!ref.evaluate) {
140 result.reason = ref.reason;
141 return result;
142 }
143 Real measure = 0;
144 for (auto const& block : snapshots) {
145 if (!units::finite(block.time) || block.time != result.time) throw std::invalid_argument("Analytic comparison needs synchronized finite times");
146 Real const volume = units::value(block.layout.cellMeasure(block.cellWidth));
147 auto const fields = sample(block, c, ref);
148 if (result.fields.empty())
149 for (auto const& field : fields)
150 result.fields.push_back(ErrorNorm{field.name, field.units});
151 result.cells += block.layout.interiorCellCount();
152 measure += volume * block.layout.interiorCellCount();
153 for (std::size_t f = 0; f < fields.size(); ++f) {
154 auto& error = result.fields[f];
155 for (std::size_t i = 0; i < fields[f].exact.size(); ++i) {
156 Real const e = abs(fields[f].numerical[i] - fields[f].exact[i]), r = abs(fields[f].exact[i]);
157 error.l1 += volume * e;
158 error.l2 += volume * e * e;
159 error.linf = std::max(error.linf, e);
160 error.referenceL1 += volume * r;
161 error.referenceL2 += volume * r * r;
162 error.referenceLinf = std::max(error.referenceLinf, r);
163 }
164 }
165 }
166 result.totalCells = result.cells;
167 for (auto& error : result.fields) {
168 error.l1 /= measure;
169 error.l2 = sqrt(error.l2 / measure);
170 error.referenceL1 /= measure;
171 error.referenceL2 = sqrt(error.referenceL2 / measure);
172 }
173 return result;
174}

177void Comparison::enforce(Config const& c) const {
178 if (c.verification.relativeL1Tolerance < 0) return;
179 if (status != "available") throw std::runtime_error("Analytic accuracy gate requires an available reference");
180 for (auto const& f : fields) {
181 bool const failed = f.referenceL1 > 0 ? f.l1 / f.referenceL1 > c.verification.relativeL1Tolerance : f.linf > c.verification.absoluteTolerance;
182 if (failed) throw std::runtime_error("Analytic accuracy tolerance exceeded: " + f.name);
183 }
