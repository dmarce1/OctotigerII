14std::vector<std::size_t> directTargets(std::size_t population, std::size_t count, std::uint64_t seed) {
15 count = std::min(count, population);
16 if (count == population) {
17 std::vector<std::size_t> result(count);
18 std::iota(result.begin(), result.end(), 0);
19 return result;
20 }
21 std::mt19937_64 generator(seed);
22 auto bounded = [&](std::uint64_t bound) {
23 // Unsigned wraparound gives 2^64 modulo bound. Reject that many values
24 // before taking the remainder, avoiding modulo bias.
25 auto const threshold = (std::uint64_t(0) - bound) % bound;
26 std::uint64_t value;
27 do {
28 value = generator();
29 } while (value < threshold);
30 return value % bound;
31 };
32 // Floyd's sampling algorithm; memory is proportional to the sample size.
33 std::set<std::size_t> selected;
34 for (std::size_t j = population - count; j < population; ++j) {
35 auto const candidate = static_cast<std::size_t>(bounded(j + 1));
36 if (!selected.insert(candidate).second) selected.insert(j);
37 }
38 return {selected.begin(), selected.end()};
39}

42ErrorNorm directErrorNorm(Field const& field, std::size_t population) {
43 using std::abs;
44 using std::isfinite;
45 using std::sqrt;
46
47 auto const count = field.exact.size();
48 if (count == 0 || count > population || count != field.numerical.size()) throw std::invalid_argument("Invalid direct norm sample");
49 ErrorNorm result{field.name, field.units};
50 for (std::size_t i = 0; i < count; ++i) {
51 Real const e = abs(field.numerical[i] - field.exact[i]), r = abs(field.exact[i]);
52 if (!isfinite(e) || !isfinite(r)) throw std::runtime_error("Nonfinite direct gravity comparison");
53 result.l1 += e;
54 result.l2 += e * e;
55 result.linf = std::max(result.linf, e);
56 result.referenceL1 += r;
57 result.referenceL2 += r * r;
58 result.referenceLinf = std::max(result.referenceLinf, r);
59 }
60 result.l1 /= count;
61 result.l2 = sqrt(result.l2 / count);
62 result.referenceL1 /= count;
63 result.referenceL2 = sqrt(result.referenceL2 / count);
64 if (count == population) {
65 result.samplingUncertaintyAvailable = true;
66 return result;
67 }
68 if (count < 2 || result.referenceL1 == 0 || result.referenceL2 == 0) {
69 result.samplingWarning = true;
70 return result;
71 }
72 Real const l1 = result.l1 / result.referenceL1, l2 = result.l2 / result.referenceL2;
73 Real variance1 = 0, variance2 = 0;
74 for (std::size_t i = 0; i < count; ++i) {
75 Real const e = abs(field.numerical[i] - field.exact[i]), r = abs(field.exact[i]);
76 Real const residual1 = (e - l1 * r) / result.referenceL1;
77 Real const residual2 = (e * e - l2 * l2 * r * r) / (result.referenceL2 * result.referenceL2);
78 variance1 += residual1 * residual1;
79 variance2 += residual2 * residual2;
80 }
81 // SRS without replacement: Var(mean) = (1-k/N) * s^2/k.
82 Real const factor = (1 - Real(count) / Real(population)) / (Real(count) * Real(count - 1));
83 result.relativeL1HalfWidth95 = 1.96 * sqrt(factor * variance1);
84 result.relativeL2HalfWidth95 = l2 > 0 ? 1.96 * sqrt(factor * variance2) / (2 * l2) : 0;
85 result.samplingUncertaintyAvailable = true;
86 result.samplingWarning = result.relativeL1HalfWidth95 > l1 || result.relativeL2HalfWidth95 > l2;
87 return result;
88}
