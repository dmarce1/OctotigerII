116 template <typename T>
117 std::shared_ptr<std::vector<T>> allocation(FieldId id, unsigned bank, std::size_t begin, std::size_t count) const {
118 std::shared_ptr<Column<T>> column;
119 {
120 std::lock_guard lock(mutex_);
121 column = std::dynamic_pointer_cast<Column<T>>(columns_.at(id));
122 }
123 if (!column) throw std::invalid_argument("Field quantity type mismatch");
124 auto data = column->banks.at(bank);
125 if (begin > data->size() || count > data->size() - begin) throw std::out_of_range("Field range exceeds partition capacity");
126 return data;
127 }
128
129 mutable std::mutex mutex_;
130 std::map<FieldId, std::shared_ptr<ColumnBase>> columns_;
