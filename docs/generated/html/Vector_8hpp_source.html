200 // Every caller writes component i and reads only that component or a
201 // scalar snapshot. Self-aliasing therefore has no loop-carried dependence.

203 template <typename Function>
204 static constexpr void forEachComponent(Function const& function) {
205 if constexpr (std::is_arithmetic_v<T>) {
206#if defined(__clang__)

212 function(i);
213 }
