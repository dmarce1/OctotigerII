54Solution solve(std::vector<units::Density> const& density, int n, units::Length cellWidth, int order, Real theta) {
55 using std::hypot;
56 using std::isfinite;
57 using std::sqrt;
58
59 // The diagonal expansion uses dimensionless coefficients, normalized to
60 // one gram and one centimeter. Restore dimensions before applying G.
61 auto const cm = units::Length::from_value(1);
62 auto const gram = units::Mass::from_value(1);
63 Real const h = cellWidth / cm;
64 if (n < 2 || (n & (n - 1)) || density.size() != static_cast<std::size_t>(n) * n * n || !(h > 0) || !isfinite(h) || order < 1 || order > 10 ||
65 !(theta > 0 && theta < 1 / sqrt(3.0)))
66 throw std::invalid_argument("Gravity requires a power-of-two cubic mesh, p=1..10, 0<theta<1/sqrt(3)");
67 std::vector<Level> levels;
68 for (int count = 1; count <= n; count *= 2)
69 levels.emplace_back(count, h * (n / count), diagonal::coefficientCount(order));
70 auto& leaves = levels.back();
71 auto const volume = cellWidth * cellWidth * cellWidth;
72 for (std::size_t i = 0; i < density.size(); ++i) {
73 if (!(density[i] >= units::Density{}) || !units::finite(density[i]) || !units::finite(density[i] * volume))
74 throw std::invalid_argument("Gravity density/mass must be finite and nonnegative");
75 leaves.moments[i][0] = density[i] * volume / gram;
76 }
77 // Upward pass: compact moments about geometric cell centers.
78 for (int level = static_cast<int>(levels.size()) - 2; level >= 0; --level) {
79 auto& parent = levels[level];
80 auto const& fine = levels[level + 1];
81 for (int z = 0; z < parent.count; ++z)
82 for (int y = 0; y < parent.count; ++y)
83 for (int x = 0; x < parent.count; ++x) {
84 Coordinate const c{x, y, z};
85 for (int slot = 0; slot < 8; ++slot)
86 add(parent.moments[index(c, parent.count)],
87 diagonal::shiftMultipole(fine.moments[index(child(c, slot), fine.count)], childOffset(slot), 0.5, order));
88 }
89 }
90 Solution result;
91 // Symmetric cell-pair traversal. Both cells always have the same width.
92 // A pair is processed at its first accepted level; descendants then see
93 // only the translated local, so there is no double counting.
94 std::function<void(int, Coordinate, Coordinate)> interact;
95 interact = [&](int depth, Coordinate a, Coordinate b) {
96 auto& level = levels[depth];
97 auto const ia = index(a, level.count), ib = index(b, level.count);
98 bool const same = a == b;
99 diagonal::Offset r{a[0] - b[0], a[1] - b[1], a[2] - b[2]};
100 Real const distance = hypot(Real(r[0]), Real(r[1]), Real(r[2]));
101 bool const leaf = depth + 1 == static_cast<int>(levels.size());
102 if (!same && !leaf && distance * theta > 1) {
103 diagonal::getOperator(order, r)->add(level.locals[ia], level.moments[ib], level.width);
104 for (auto& value : r)
105 value = -value;
106 diagonal::getOperator(order, r)->add(level.locals[ib], level.moments[ia], level.width);
107 ++result.statistics.multipolePairs;
108 } else if (leaf) {
109 if (same) return;
110 diagonal::addDirect(level.locals[ia], level.moments[ib][0], r, level.width);
111 for (auto& value : r)
112 value = -value;
113 diagonal::addDirect(level.locals[ib], level.moments[ia][0], r, level.width);
114 ++result.statistics.directPairs;
115 } else {
116 for (int i = 0; i < 8; ++i)
117 for (int j = (same ? i : 0); j < 8; ++j)
118 interact(depth + 1, child(a, i), child(b, j));
119 }
120 };
121 interact(0, {0, 0, 0}, {0, 0, 0});
122 // Downward pass: normalized derivatives scale with the level width.
123 for (std::size_t depth = 0; depth + 1 < levels.size(); ++depth) {
124 auto const& parent = levels[depth];
125 auto& fine = levels[depth + 1];
126 for (int z = 0; z < parent.count; ++z)
127 for (int y = 0; y < parent.count; ++y)
128 for (int x = 0; x < parent.count; ++x) {
129 Coordinate const c{x, y, z};
130 for (int slot = 0; slot < 8; ++slot)
131 add(fine.locals[index(child(c, slot), fine.count)],
132 diagonal::shiftLocal(parent.locals[index(c, parent.count)], childOffset(slot), 0.5, order));
133 }
134 }
135 result.fields.resize(density.size());
136 for (std::size_t i = 0; i < density.size(); ++i) {
137 auto& field = result.fields[i];
138 auto const& local = leaves.locals[i];
139 field.potential() = constants::G * (gram / cm) * local[0];
140 field.acceleration(0) = -constants::G * (gram / cm) * diagonal::derivative(local, 1, 0, 0) / cellWidth;
141 field.acceleration(1) = -constants::G * (gram / cm) * diagonal::derivative(local, 0, 1, 0) / cellWidth;
142 field.acceleration(2) = -constants::G * (gram / cm) * diagonal::derivative(local, 0, 0, 1) / cellWidth;
143 if (!finite(field)) throw std::runtime_error("Nonfinite gravity solution");
144 }
145 return result;
146}
