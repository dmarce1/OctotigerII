72 static Waves physicalFlux(State const& u, int normal, units::Velocity chat) {
73 using std::sqrt;
74
75 FpeGuard guard;
76 checkState(u);
77 if (normal < 0 || normal >= ndim || (!(chat > units::Velocity{}) || !units::finite(chat))) throw std::invalid_argument("Invalid M1 normal or speed");
78 std::array<Real, ndim> f{};
79 if (u[0] > units::EnergyDensity{})
80 for (int i = 0; i < ndim; ++i)
81 f[i] = u[i + 1] / u[0];
82 Real f2 = 0;
83 for (auto component : f)
84 f2 += component * component;
85 if (f2 > 1) {
86 for (auto& x : f)
87 x /= sqrt(f2);
88 f2 = 1;
89 }
90 Real fn = f[normal], mu2 = f2 > 0 ? std::min(Real(1), fn * fn / f2) : 0;
91 Real s = sqrt(4 - 3 * f2);
92 auto isotropic = u[0] * ((1 - f2) / (s + 1));
93 auto directed = u[0] * (3 / (s + 2));
94 Real radical = sqrt(std::max(Real(0), 2 * (1 - f2) * (s + mu2 * (s - 2)) / (s + 1)));
95 Waves out;
96 out.minus = chat * ((fn - radical) / s);
97 out.plus = chat * ((fn + radical) / s);
98 out.flux[0] = chat * u[0] * fn;
99 for (int i = 0; i < ndim; ++i)
100 out.flux[i + 1] = chat * (directed * fn * f[i] + (normal == i ? isotropic : units::Pressure{}));
101 return out;
102 }

106 static Flux hll(State const& left, State const& right, int normal, units::Velocity chat) {
107 FpeGuard guard;
108 auto ul = canonical(left), ur = canonical(right);
109 auto l = physicalFlux(ul, normal, chat), r = physicalFlux(ur, normal, chat);
110 auto sm = std::min(l.minus, r.minus), sp = std::max(l.plus, r.plus);
111 if (sm >= units::Velocity{}) return l.flux;
112 if (sp <= units::Velocity{}) return r.flux;
113 Real wr = sp / (sp - sm), wl = -sm / (sp - sm);
114 return wr * (l.flux - sm * ul) + wl * (r.flux - sp * ur);
115 }

118 static State roundoffState(State u, State const& updateScale) {
119 auto scale = updateScale[0];
120 for (int i = 1; i < State::size(); ++i)
121 scale = std::max(scale, updateScale[i]);
122 auto tolerance = roundoff * scale, norm = magnitude(u);
123 if (u[0] < units::EnergyDensity{} && -u[0] <= tolerance && norm <= tolerance) return {};
124 if (u[0] >= units::EnergyDensity{} && norm > u[0] && norm - u[0] <= tolerance)
125 for (int i = 1; i < State::size(); ++i)
126 u[i] *= Real(u[0] / norm);
127 checkState(u);
128 return u;
129 }
