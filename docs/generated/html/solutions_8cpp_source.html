10Reference sodReference(Config const& c) {
11 using std::pow;
12 using std::sqrt;
13
14 Real const gamma = c.hydro.gamma;
15 auto const rhoL = units::Density::from_value(1), rhoR = units::Density::from_value(0.125);
16 auto const pL = units::Pressure::from_value(1), pR = units::Pressure::from_value(0.1);
17 auto const aL = units::sqrt(gamma * pL / rhoL), aR = units::sqrt(gamma * pR / rhoR);
18 // Match the left rarefaction and right shock curves. Bisection brackets the
19 // unique star pressure for these Sod states for any supported gamma > 1.
20 auto wave = [gamma](units::Pressure p, units::Pressure p0, units::Density rho, units::Velocity a) {
21 if (p > p0) return (p - p0) * units::sqrt((2.0 / (gamma + 1) / rho) / (p + (gamma - 1) / (gamma + 1) * p0));
22 return 2.0 * a / (gamma - 1) * (pow(Real(p / p0), (gamma - 1) / (2 * gamma)) - 1);
23 };
24 auto lo = pR, hi = pL;
25 for (int iteration = 0; iteration < 100; ++iteration) {
26 auto const p = (lo + hi) / 2.0;
27 if (wave(p, pL, rhoL, aL) + wave(p, pR, rhoR, aR) > units::Velocity{})
28 hi = p;
29 else
30 lo = p;
31 }
32 auto const pStar = (lo + hi) / 2.0;
33 auto const uStar = (wave(pStar, pR, rhoR, aR) - wave(pStar, pL, rhoL, aL)) / 2.0;
34 Real const ratioL = pStar / pL, ratioR = pStar / pR;
35 auto const rhoStarL = rhoL * pow(ratioL, 1 / gamma);
36 Real const beta = (gamma - 1) / (gamma + 1);
37 auto const rhoStarR = rhoR * (ratioR + beta) / (beta * ratioR + 1);
38 auto const aStarL = aL * pow(ratioL, (gamma - 1) / (2 * gamma));
39 auto const shock = aR * sqrt((gamma + 1) / (2 * gamma) * ratioR + (gamma - 1) / (2 * gamma));
40 auto const middle = (c.mesh.lower + c.mesh.upper) / 2.0;
41 Reference result;
42 result.name = "Sod exact Riemann solution";
43 result.reason = "Sod reference is valid only before the first wave reaches a domain boundary";
44 result.validUntil = (c.mesh.upper - c.mesh.lower) / (2.0 * std::max(aL, shock));
45 result.evaluate = [=](mesh::PhysicalCoordinates const& position, units::Time time) {
46 using std::pow;
47
48 ExactState state;
49 auto& q = state.hydro;
50 if (time == units::Time{}) {
51 q.density() = position[0] < middle ? rhoL : rhoR;
52 q.pressure() = position[0] < middle ? pL : pR;
53 return state;
54 }
55 auto const xi = (position[0] - middle) / time;
56 if (xi <= -aL) {
57 q.density() = rhoL;
58 q.pressure() = pL;
59 } else if (xi < uStar - aStarL) {
60 auto const velocity = 2.0 / (gamma + 1) * (aL + xi);
61 auto const sound = 2.0 / (gamma + 1) * (aL - (gamma - 1) / 2.0 * xi);
62 q.density() = rhoL * pow(Real(sound / aL), 2 / (gamma - 1));
63 q.pressure() = pL * pow(Real(sound / aL), 2 * gamma / (gamma - 1));
64 q.velocity(0) = velocity;
65 } else if (xi < uStar) {
66 q.density() = rhoStarL;
67 q.pressure() = pStar;
68 q.velocity(0) = uStar;
69 } else if (xi < shock) {
70 q.density() = rhoStarR;
71 q.pressure() = pStar;
72 q.velocity(0) = uStar;
73 } else {
74 q.density() = rhoR;
75 q.pressure() = pR;
76 }
77 return state;
78 };
79 return result;
80}

83Reference sphereReference(Config const& c, bool gaussian) {
84 using std::erf;
85 using std::exp;
86 using std::expm1;
87 using std::sqrt;
88
89 auto const length = c.mesh.upper - c.mesh.lower;
90 auto const center = (c.mesh.lower + c.mesh.upper) / 2.0;
91 auto const radius = (gaussian ? 0.5 : 0.25) * length;
92 auto const sigma = 0.15 * length;
93 auto const density = units::Density::from_value(1e4);
94 Real constexpr pi = std::numbers::pi_v<Real>;
95 // Shell theorem: phi(r)=-G[M(<r)/r + 4 pi integral_r^R rho(s) s ds].
96 // A series avoids cancellation in the enclosed Gaussian mass near r=0.
97 auto mass = [=](units::Length r) -> units::Mass {
98 using std::erf;
99 using std::exp;
100 using std::sqrt;
101
102 if (!gaussian) return (4 * pi / 3) * density * r * r * r;
103 Real const q = r / sigma, q2 = q * q;
104 Real const integral =
105 q < 0.01 ? q * q2 * (1.0 / 3 - q2 / 10 + q2 * q2 / 56 - q2 * q2 * q2 / 432) : sqrt(pi / 2) * erf(q / sqrt(2.0)) - q * exp(-q2 / 2);
106 return 4 * pi * density * sigma * sigma * sigma * integral;
107 };
108 Reference result;
109 result.name = gaussian ? "Spherically truncated Gaussian gravity" : "Uniform sphere gravity";
110 result.evaluate = [=](mesh::PhysicalCoordinates const& position, units::Time) {
111 using std::exp;
112 using std::expm1;
113
114 ExactState state;
115 units::Length r{};
116 for (auto const& x : position)
117 r = units::hypot(r, x - center);
118 auto const enclosed = mass(std::min(r, radius));
119 if (r >= radius)
120 state.gravity.potential() = -constants::G * enclosed / r;
121 else {
122 units::Quantity<-1, 1, 0> shells{};
123 if (gaussian) {
124 Real const q = r / sigma, cutoff = radius / sigma;
125 shells = 4 * pi * density * sigma * sigma * exp(-q * q / 2) * (-expm1(-(cutoff * cutoff - q * q) / 2));
126 } else
127 shells = 2 * pi * density * (radius * radius - r * r);
128 state.gravity.potential() = -constants::G * (shells + (r == units::Length{} ? units::Quantity<-1, 1, 0>{} : enclosed / r));
129 }
130 if (r != units::Length{})
131 for (int axis = 0; axis < ndim; ++axis)
132 state.gravity.acceleration(axis) = -constants::G * enclosed / (r * r) * ((position[axis] - center) / r);
133 return state;
134 };
135 return result;
136}
