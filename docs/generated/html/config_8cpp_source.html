66 char const* selectedKey(po::variables_map const& values, char const* key, char const* legacy = nullptr) {
67 bool const canonicalPresent = values.count(key) != 0;
68 bool const legacyPresent = legacy != nullptr && values.count(legacy) != 0;
69 if (canonicalPresent && legacyPresent) throw std::invalid_argument(std::string("Conflicting names for ") + key + ": " + legacy);
70 if (canonicalPresent) return key;
71 if (legacyPresent) return legacy;
72 return nullptr;
73 }

141void Config::validate() const {
142 using std::isfinite;
143 using std::sqrt;
144
145 if (verification.analytic != "auto" && verification.analytic != "on" && verification.analytic != "off")
146 throw std::invalid_argument("verification.analytic must be auto, on, or off");
147 if (!isfinite(verification.relativeL1Tolerance) || !isfinite(verification.absoluteTolerance) ||
148 (verification.relativeL1Tolerance < 0 && verification.relativeL1Tolerance != -1) || verification.absoluteTolerance < 0 ||
149 (verification.analytic == "off" && verification.relativeL1Tolerance >= 0))
150 throw std::invalid_argument("Invalid analytic verification tolerance or disabled accuracy gate");
151 if (verification.gravityReference != "direct" && verification.gravityReference != "continuum")
152 throw std::invalid_argument("verification.gravityReference must be direct or continuum");
153 if (verification.directMaxPairs < 1 || verification.directSamples < 0)
154 throw std::invalid_argument("Direct pair budget must be positive and directSamples nonnegative");
155 if (randomSeed < 0) throw std::invalid_argument("randomSeed must be nonnegative");
156 validateProblem(*this);
157 if (mesh.cells < 4 || mesh.cells > 128 || (mesh.cells & (mesh.cells - 1)) || mesh.level < 0 || mesh.level > 6)
158 throw std::invalid_argument("mesh: cells=power of two in [4,128], level=0..6");
159 if (!(mesh.upper > mesh.lower) || !units::finite(mesh.upper - mesh.lower) || !(runtime.stopTime >= units::Time{}) ||
160 !(timestep.cfl > 0 && timestep.cfl <= 0.5) || !(hydro.gamma > 1) || !(radiation.lightSpeedRatio > 0 && radiation.lightSpeedRatio <= 1) ||
161 runtime.maxSteps < 1 || output.every < 1 || runtime.workerTasks < 0)
162 throw std::invalid_argument("Invalid domain, timestep, gas, radiation, or output setting");
163 if (gravityEnabled() && mesh.periodic) throw std::invalid_argument("Gravity currently requires 3D isolated boundaries");
164 if (gravity.multipoleOrder < 1 || gravity.multipoleOrder > 10 || !(gravity.openingAngle > 0 && gravity.openingAngle < 1 / sqrt(3.0)))
165 throw std::invalid_argument("Gravity requires order 1..10 and 0<openingAngle<1/sqrt(3)");
166 if (output.directory.empty()) throw std::invalid_argument("Empty output directory");
167}

169Config parseConfig(std::vector<std::string> const& arguments) {
170 po::options_description settings("Simulation options");
171 addSettings(settings);
172 po::options_description legacy("Legacy aliases");
173 addLegacySettings(legacy);
174 po::options_description buildChoices("Build-time choices");
175 buildChoices.add_options()("problem.name", po::value<std::string>(), "Select OCTOTIGERII_PROBLEM in CMake")(
176 "mesh.ndim", po::value<int>(), "Select OCTOTIGERII_NDIM in CMake");
177 po::options_description iniOptions;
178 iniOptions.add(settings).add(legacy).add(buildChoices);
179 po::options_description commandOptions;
180 commandOptions.add(iniOptions);
181 commandOptions.add_options()("config", po::value<std::vector<std::string>>()->composing(), "INI file (repeatable)");
182
183 po::variables_map commandValues;
184 po::store(po::command_line_parser(arguments)
185 .options(commandOptions)
186 .style(po::command_line_style::allow_long | po::command_line_style::long_allow_adjacent)
187 .run(),
188 commandValues);
189 po::notify(commandValues);
190
191 Config config;
192 problemDefaults(config);
193 if (commandValues.count("config")) {
194 for (std::string const& path : commandValues["config"].as<std::vector<std::string>>()) {
195 std::ifstream input(path);
196 if (!input) throw std::runtime_error("Cannot open config: " + path);
197 po::variables_map iniValues;
198 po::store(po::parse_config_file(input, iniOptions), iniValues);
199 po::notify(iniValues);
200 applySettings(config, iniValues);
201 }
202 }
203 applySettings(config, commandValues);
204 config.validate();
205 return config;
206}
